void add_waypoint(
    HovercraftRace *race,
    double x,
    double y,
    RaceSurface surface
) {
    if (race->waypoint_count >=
        MAX_WAYPOINTS)
        return;

    RaceWaypoint *w =
        &race->waypoints[
            race->waypoint_count++
        ];

    w->id =
        race->waypoint_count - 1;

    w->x = x;
    w->y = y;

    w->surface = surface;

    w->width = 40.0;

    switch (surface) {

        case SURFACE_WATER:
            w->recommended_speed = 30.0;
            break;

        case SURFACE_MARSH:
            w->recommended_speed = 23.0;
            break;

        case SURFACE_MUD:
            w->recommended_speed = 18.0;
            break;

        case SURFACE_SAWGRASS:
            w->recommended_speed = 15.0;
            break;

        default:
            w->recommended_speed = 25.0;
    }
}

AI racing line

The AI looks at the next waypoint and adjusts its speed.

double calculate_target_speed(
    Racer *racer,
    RaceWaypoint *waypoint
) {
    double skill_bonus =
        racer->skill * 8.0;

    double target =
        waypoint->recommended_speed +
        skill_bonus;

    if (target >
        racer->vehicle->maximum_speed)

        target =
            racer->vehicle->maximum_speed;

    return target;
}
Steering toward the waypoint
double angle_to_waypoint(
    Hovercraft *vehicle,
    RaceWaypoint *waypoint
) {
    double dx =
        waypoint->x -
        vehicle->position.x;

    double dy =
        waypoint->y -
        vehicle->position.y;

    double target =
        atan2(dy, dx);

    double difference =
        target -
        vehicle->position.heading;

    while (difference > M_PI)
        difference -=
            2.0 * M_PI;

    while (difference < -M_PI)
        difference +=
            2.0 * M_PI;

    return difference;
}

AI steering:

double racing_ai_steering(
    Racer *racer,
    RaceWaypoint *waypoint
) {
    double angle =
        angle_to_waypoint(
            racer->vehicle,
            waypoint
        );

    double response =
        1.0 +
        racer->skill;

    return
        angle * response;
}
Race positioning
double distance_to_waypoint(
    Hovercraft *vehicle,
    RaceWaypoint *waypoint
) {
    double dx =
        waypoint->x -
        vehicle->position.x;

    double dy =
        waypoint->y -
        vehicle->position.y;

    return sqrt(
        dx * dx +
        dy * dy
    );
}
void check_waypoint(
    Racer *racer,
    HovercraftRace *race
) {
    RaceWaypoint *waypoint =
        &race->waypoints[
            racer->current_waypoint
        ];

    double distance =
        distance_to_waypoint(
            racer->vehicle,
            waypoint
        );

    if (distance < waypoint->width) {

        racer->current_waypoint++;

        if (racer->current_waypoint >=
            race->waypoint_count) {

            racer->current_waypoint = 0;

            racer->current_lap++;

            if (racer->current_lap >=
                race->laps) {

                racer->finished = true;
            }
        }
    }
}

    RaceWeather *weather

        0.001;

void apply_race_stress(
    Racer *racer,
    double throttle,
    double delta_time
) {
    double stress =
        throttle *
        delta_time *
        0.01;

    racer->damage +=
        stress;

    /*
     * Poor-condition vehicles
     * suffer more.
     */

    if (racer->vehicle->condition < 0.5) {

        racer->damage +=

        racer->finished = false;

Aerodynamic downforce

The important characteristic is that downforce increases rapidly with speed.

double calculate_downforce(
    FerrariF1 *car,
    double speed
) {
    /*
     * Simplified game approximation.
     *
     * Downforce ∝ velocity²
     */

    double normalized =
        speed / car->top_speed;

    return
        car->downforce *
        normalized *
        normalized;
}

At low speed:

LOW SPEED
   ↓
LOW DOWNFORCE
   ↓
LIMITED GRIP

At high speed:

HIGH SPEED
   ↓
HIGH DOWNFORCE
   ↓
EXTREME CORNERING

That gives the car its characteristic F1-style behaviour.

Total grip
double calculate_grip(
    FerrariF1 *car,
    VehiclePhysics *physics
) {
    double downforce =
        calculate_downforce(
            car,
            physics->speed
        );

    double aerodynamic_grip =
        downforce / 1000.0;

    double mechanical_grip =
        car->grip;

    return
        mechanical_grip +
        aerodynamic_grip;
}
Acceleration
double calculate_acceleration(
    FerrariF1 *car,
    VehiclePhysics *physics
) {
    double combustion =
        car->engine_power *
        physics->throttle;

    double electric =
        car->electric_power *
        physics->throttle *
        (car->battery / car->battery_capacity);

    double total_power =
        combustion +
        electric;

    double acceleration =
        total_power /
        car->mass /
        fmax(
            physics->speed,
            1.0
        );

    /*
     * Prevent unrealistic acceleration.
     */

    if (acceleration >
        car->acceleration)

        acceleration =
            car->acceleration;

    return acceleration;
}
Energy deployment

Give the player control over the hybrid system.

typedef enum {

    ENERGY_BALANCED,
    ENERGY_ATTACK,
    ENERGY_DEFEND,
    ENERGY_RECHARGE

} EnergyMode;
void update_energy(
    FerrariF1 *car,
    EnergyMode mode,
    double delta_time
) {
    double usage = 0.0;

    switch (mode) {

        case ENERGY_ATTACK:

            usage = 4.0;
            break;

        case ENERGY_DEFEND:

            usage = 2.0;
            break;

        case ENERGY_BALANCED:

            usage = 1.5;
            break;

        case ENERGY_RECHARGE:

            usage = -1.0;
            break;
    }

    car->battery -=
        usage * delta_time;

    if (car->battery < 0)
        car->battery = 0;

    if (car->battery >
        car->battery_capacity)

        car->battery =
            car->battery_capacity;
}

Update physics
void update_ferrari_f1(
    FerrariF1 *car,
    VehiclePhysics *physics,
    EnergyMode energy_mode,
    double delta_time
) {
    if (!car->active)
        return;

    double acceleration =
        calculate_acceleration(
            car,
            physics
        );

    double braking =
        calculate_braking(
            car,
            physics
        );

    double drag =
        calculate_drag(
            car,
            physics->speed
        );

    double net_acceleration =
        acceleration -
        braking -
        drag / car->mass;

    physics->speed +=
        net_acceleration *
        delta_time;

    if (physics->speed < 0)
        physics->speed = 0;

    if (physics->speed >
        car->top_speed)

        physics->speed =
            car->top_speed;

    /*
     * Position integration.
     */

    physics->velocity_x =
        cos(physics->heading) *
        physics->speed;

    physics->velocity_y =
        sin(physics->heading) *
        physics->speed;

    physics->x +=
        physics->velocity_x *
        delta_time;

    physics->y +=
        physics->velocity_y *
        delta_time;

    apply_steering(
        car,
        physics,
        delta_time
    );

    update_energy(
        car,
        energy_mode,
        delta_time
    );
}

double tyre_grip(
    FerrariF1 *car,
    TyreCompound compound
) {
    double grip = 1.0;

    switch (compound) {

        case TYRE_SOFT:
            grip = 1.20;
            break;

        case TYRE_MEDIUM:
            grip = 1.10;
            break;

        case TYRE_HARD:
            grip = 1.00;
            break;

        case TYRE_INTERMEDIATE:
            grip = 0.85;
            break;

        case TYRE_WET:
            grip = 0.70;
            break;
    }

    /*
     * Temperature penalty.
     */

    double temperature_error =
        fabs(
            car->tyre_temperature -
            90.0
        );

    grip -=
        temperature_error *
        0.002;

    return fmax(
        grip,
        0.40
    );
}

Passengers move around the ship.

void passenger_update(
    Passenger *p,
    CruiseShip *ship,
    double delta_time
) {
    if (!p->aboard)
        return;

    /*
     * Passengers spend money
     * while exploring the ship.
     */

    p->money -=
        p->spending_rate *
        delta_time;

    if (p->money < 0)
        p->money = 0;

    /*
     * Satisfaction gradually
     * changes during the voyage.
     */

    p->satisfaction +=
        0.001 *
        delta_time;

    if (p->satisfaction > 1.0)
        p->satisfaction = 1.0;
}

6. Docking
double distance_to_port(
    CruiseShip *ship,
    CruisePort *port
) {
    double dx =
        ship->x - port->x;

    double dy =
        ship->y - port->y;

    return sqrt(
        dx * dx +
        dy * dy
    );
}
bool ship_can_dock(
    CruiseShip *ship,
    CruisePort *port
) {
    return
        distance_to_port(ship, port)
        <= port->docking_radius;
}
7. Departure
void depart_port(
    CruiseShip *ship

    if (ship->current_passengers <= 0)

    ship->boarding = false;
    ship->disembarking = false;

8. International-water system

For the game, create a Maritime Zone System.

typedef enum {

    ZONE_PORT,
    ZONE_TERRITORIAL,
    ZONE_INTERNATIONAL,
    ZONE_DESTINATION

} MaritimeZone;

The world determines the current zone:

MaritimeZone get_maritime_zone(
    double x,
    double y
) {
    /*
     * Simplified game-world boundary.
     *
     * Real-world maritime boundaries are
     * much more complicated and should not
     * be treated as exact legal boundaries.
     */

    double distance_from_coast =
        sqrt(
            x * x +
            y * y
        );

    if (distance_from_coast < 10.0)
        return ZONE_PORT;

    if (distance_from_coast < 50.0)
        return ZONE_TERRITORIAL;

    return ZONE_INTERNATIONAL;
}
9. Rules change at the international line
typedef struct {

    bool normal_port_rules;

    bool international_navigation;

    bool open_ocean_events;

    bool special_missions;

    double encounter_probability;

    double weather_probability;

} MaritimeRules;
MaritimeRules get_maritime_rules(
    MaritimeZone zone
) {
    MaritimeRules rules = {0};

    switch (zone) {

        case ZONE_PORT:

            rules.normal_port_rules = true;
            rules.international_navigation = false;
            rules.open_ocean_events = false;

            break;

        case ZONE_TERRITORIAL:

            rules.normal_port_rules = true;
            rules.international_navigation = true;
            rules.open_ocean_events = false;

            break;

        case ZONE_INTERNATIONAL:

            rules.normal_port_rules = false;
            rules.international_navigation = true;
            rules.open_ocean_events = true;

            rules.special_missions = true;

            rules.encounter_probability = 0.10;
            rules.weather_probability = 0.20;

            break;

        case ZONE_DESTINATION:

            rules.normal_port_rules = true;

            break;
    }

    return rules;
}

10. Crossing the boundary
void check_maritime_transition(
    CruiseShip *ship,
    MaritimeZone previous,
    MaritimeZone current
) {
    if (
        previous != ZONE_INTERNATIONAL &&
        current == ZONE_INTERNATIONAL
    ) {

        printf(
            "%s has entered international waters.\n",
            ship->callsign
        );

        printf(
            "Open-ocean game systems enabled.\n"
        );
    }

    if (
        previous == ZONE_INTERNATIONAL &&
        current != ZONE_INTERNATIONAL
    ) {

        printf(
            "%s has left international waters.\n",
            ship->callsign
        );
    }
}
11. Open-ocean events

Once the ship crosses the line, the game can activate a different event pool.

typedef enum {

    EVENT_NONE,
    EVENT_STORM,
    EVENT_MECHANICAL,
    EVENT_WILDLIFE,
    EVENT_OTHER_SHIP,
    EVENT_SUNSET,
    EVENT_NAVIGATION,
    EVENT_PARTY,
    EVENT_RARE_ENCOUNTER

} OceanEvent;
OceanEvent generate_ocean_event(
    MaritimeRules *rules
) {
    if (!rules->open_ocean_events)
        return EVENT_NONE;

    double r =
        (double)rand() /
        RAND_MAX;

    if (r < 0.02)
        return EVENT_STORM;

    if (r < 0.04)
        return EVENT_MECHANICAL;

    if (r < 0.07)
        return EVENT_OTHER_SHIP;

    if (r < 0.10)
        return EVENT_PARTY;

    return EVENT_NONE;
}

/*
 * ============================================================
 * CAPTAIN HAROLD VANE
 * RUTHLESS TICKET INSPECTOR
 * ============================================================
 *
 * Fictional game NPC.
 *
 * Captain Vane operates aboard:
 *     TITANIC
 *     OLYMPIA
 *     AEROKRAFT AIRSHIPS
 *
 * He is obsessed with ticket enforcement.
 *
 * Gameplay:
 *     - Ticket inspection
 *     - Passenger identification
 *     - Suspicion
 *     - Security escalation
 *     - Restricted-area access
 *     - Removal at next safe port
 *     - Fines / reputation penalties
 *

#include <stdio.h>

Inspection
InspectionState inspect_passenger(
    Captain *captain,
    Passenger *passenger
) {
    if (!captain->captain_on_duty)
        return INSPECTION_NORMAL;

    switch (passenger->ticket) {

        case TICKET_VALID:

            return INSPECTION_NORMAL;

        case TICKET_EXPIRED:

            return INSPECTION_SUSPICIOUS;

        case TICKET_WRONG_CLASS:

            return INSPECTION_SUSPICIOUS;

        case TICKET_INVALID:

            return INSPECTION_SECURITY;

        case TICKET_MISSING:

            return INSPECTION_SECURITY;
    }

    return INSPECTION_NORMAL;
}
Captain dialogue
void captain_address_passenger(
    Captain *captain,

) {

Escalation
void escalate_ticket_case(
    Captain *captain,
    Passenger *passenger
) {
    if (
        passenger->ticket ==
        TICKET_VALID
    ) {
        passenger->state =
            INSPECTION_NORMAL;

        return;
    }

    if (
        passenger->ticket ==
        TICKET_EXPIRED ||
        passenger->ticket ==
        TICKET_WRONG_CLASS
    ) {
        passenger->state =
            INSPECTION_SUSPICIOUS;

        return;
    }

    passenger->state =
        INSPECTION_SECURITY;
}
Removal from the voyage

Rather than ejecting someone from a moving ship or airship, the captain can have them escorted off at the next safe port.

void remove_at_next_port(
    Passenger *passenger
) {
    if (
        passenger->state !=
        INSPECTION_SECURITY
    )
        return;

    printf(
        "Passenger %s will be removed "
        "at the next safe port.\n",
        passenger->name
    );

    passenger->state =
        INSPECTION_REMOVAL;
}
