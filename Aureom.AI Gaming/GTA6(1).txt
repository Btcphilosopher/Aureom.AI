Generate a circuit
void add_waypoint(
    HovercraftRace *race,
    double x,
    double y,
    RaceSurface surface
) {
    if (race->waypoint_count >=
        MAX_WAYPOINTS)
        return;

    RaceWaypoint *w =
        &race->waypoints[
            race->waypoint_count++
        ];

    w->id =
        race->waypoint_count - 1;

    w->x = x;
    w->y = y;

    w->surface = surface;

    w->width = 40.0;

    switch (surface) {

        case SURFACE_WATER:
            w->recommended_speed = 30.0;
            break;

        case SURFACE_MARSH:
            w->recommended_speed = 23.0;
            break;

        case SURFACE_MUD:
            w->recommended_speed = 18.0;
            break;

        case SURFACE_SAWGRASS:
            w->recommended_speed = 15.0;
            break;

        default:
            w->recommended_speed = 25.0;
    }
}

AI racing line

The AI looks at the next waypoint and adjusts its speed.

double calculate_target_speed(
    Racer *racer,
    RaceWaypoint *waypoint
) {
    double skill_bonus =
        racer->skill * 8.0;

    double target =
        waypoint->recommended_speed +
        skill_bonus;

    if (target >
        racer->vehicle->maximum_speed)

        target =
            racer->vehicle->maximum_speed;

    return target;
}
Steering toward the waypoint
double angle_to_waypoint(
    Hovercraft *vehicle,
    RaceWaypoint *waypoint
) {
    double dx =
        waypoint->x -
        vehicle->position.x;

    double dy =
        waypoint->y -
        vehicle->position.y;

    double target =
        atan2(dy, dx);

    double difference =
        target -
        vehicle->position.heading;

    while (difference > M_PI)
        difference -=
            2.0 * M_PI;

    while (difference < -M_PI)
        difference +=
            2.0 * M_PI;

    return difference;
}

AI steering:

double racing_ai_steering(
    Racer *racer,
    RaceWaypoint *waypoint
) {
    double angle =
        angle_to_waypoint(
            racer->vehicle,
            waypoint
        );

    double response =
        1.0 +
        racer->skill;

    return
        angle * response;
}
Race positioning
double distance_to_waypoint(
    Hovercraft *vehicle,
    RaceWaypoint *waypoint
) {
    double dx =
        waypoint->x -
        vehicle->position.x;

    double dy =
        waypoint->y -
        vehicle->position.y;

    return sqrt(
        dx * dx +
        dy * dy
    );
}
void check_waypoint(
    Racer *racer,
    HovercraftRace *race
) {
    RaceWaypoint *waypoint =
        &race->waypoints[
            racer->current_waypoint
        ];

    double distance =
        distance_to_waypoint(
            racer->vehicle,
            waypoint
        );

    if (distance < waypoint->width) {

        racer->current_waypoint++;

        if (racer->current_waypoint >=
            race->waypoint_count) {

            racer->current_waypoint = 0;

            racer->current_lap++;

            if (racer->current_lap >=
                race->laps) {

                racer->finished = true;
            }
        }
    }
}

double weather_speed_modifier(
    RaceWeather *weather
) {
    double modifier = 1.0;

    modifier -=
        weather->rain *
        0.003;

    modifier -=
        weather->wind *
        0.001;

    if (weather->storm)

    if (modifier < 0.35)
        modifier = 0.35;

void apply_race_stress(
    Racer *racer,
    double throttle,
    double delta_time
) {
    double stress =
        throttle *
        delta_time *
        0.01;

    racer->damage +=
        stress;

    /*
     * Poor-condition vehicles
     * suffer more.
     */

    if (racer->vehicle->condition < 0.5) {

        racer->damage +=
            stress * 2.0;

    if (racer->damage >= 1.0) {

        racer->retired = true;
        racer->finished = false;

}

Aerodynamic downforce

The important characteristic is that downforce increases rapidly with speed.

double calculate_downforce(
    FerrariF1 *car,
    double speed
) {
    /*
     * Simplified game approximation.
     *
     * Downforce ∝ velocity²
     */

    double normalized =
        speed / car->top_speed;

    return
        car->downforce *
        normalized *
        normalized;
}

At low speed:

LOW SPEED
   ↓
LOW DOWNFORCE
   ↓
LIMITED GRIP

At high speed:

HIGH SPEED
   ↓
HIGH DOWNFORCE
   ↓
EXTREME CORNERING

That gives the car its characteristic F1-style behaviour.

Total grip
double calculate_grip(
    FerrariF1 *car,
    VehiclePhysics *physics
) {
    double downforce =
        calculate_downforce(
            car,
            physics->speed
        );

    double aerodynamic_grip =
        downforce / 1000.0;

    double mechanical_grip =
        car->grip;

    return
        mechanical_grip +
        aerodynamic_grip;
}
Acceleration
double calculate_acceleration(
    FerrariF1 *car,
    VehiclePhysics *physics
) {
    double combustion =
        car->engine_power *
        physics->throttle;

    double electric =
        car->electric_power *
        physics->throttle *
        (car->battery / car->battery_capacity);

    double total_power =
        combustion +
        electric;

    double acceleration =
        total_power /
        car->mass /
        fmax(
            physics->speed,
            1.0
        );

    /*
     * Prevent unrealistic acceleration.
     */

    if (acceleration >
        car->acceleration)

        acceleration =
            car->acceleration;

    return acceleration;
}
Energy deployment

Give the player control over the hybrid system.

typedef enum {

    ENERGY_BALANCED,
    ENERGY_ATTACK,
    ENERGY_DEFEND,
    ENERGY_RECHARGE

} EnergyMode;
void update_energy(
    FerrariF1 *car,
    EnergyMode mode,
    double delta_time
) {
    double usage = 0.0;

    switch (mode) {

        case ENERGY_ATTACK:

            usage = 4.0;
            break;

        case ENERGY_DEFEND:

            usage = 2.0;
            break;

        case ENERGY_BALANCED:

            usage = 1.5;
            break;

        case ENERGY_RECHARGE:

            usage = -1.0;
            break;
    }

    car->battery -=
        usage * delta_time;

    if (car->battery < 0)
        car->battery = 0;

    if (car->battery >
        car->battery_capacity)

        car->battery =
            car->battery_capacity;
}

Update physics
void update_ferrari_f1(
    FerrariF1 *car,
    VehiclePhysics *physics,
    EnergyMode energy_mode,
    double delta_time
) {
    if (!car->active)
        return;

    double acceleration =
        calculate_acceleration(
            car,
            physics
        );

    double braking =
        calculate_braking(
            car,
            physics
        );

    double drag =
        calculate_drag(
            car,
            physics->speed
        );

    double net_acceleration =
        acceleration -
        braking -
        drag / car->mass;

    physics->speed +=
        net_acceleration *
        delta_time;

    if (physics->speed < 0)
        physics->speed = 0;

    if (physics->speed >
        car->top_speed)

        physics->speed =
            car->top_speed;

    /*
     * Position integration.
     */

    physics->velocity_x =
        cos(physics->heading) *
        physics->speed;

    physics->velocity_y =
        sin(physics->heading) *
        physics->speed;

    physics->x +=
        physics->velocity_x *
        delta_time;

    physics->y +=
        physics->velocity_y *
        delta_time;

    apply_steering(
        car,
        physics,
        delta_time
    );

    update_energy(
        car,
        energy_mode,
        delta_time
    );
}

typedef enum {

double tyre_grip(
    FerrariF1 *car,
    TyreCompound compound
) {
    double grip = 1.0;

    switch (compound) {

        case TYRE_SOFT:
            grip = 1.20;
            break;

        case TYRE_MEDIUM:
            grip = 1.10;
            break;

        case TYRE_HARD:
            grip = 1.00;
            break;

        case TYRE_INTERMEDIATE:
            grip = 0.85;
            break;

        case TYRE_WET:
            grip = 0.70;
            break;
    }

    /*
     * Temperature penalty.
     */

    double temperature_error =
        fabs(
            car->tyre_temperature -
            90.0
        );

    grip -=
        temperature_error *
        0.002;

    return fmax(
        grip,
        0.40
    );
}

void passenger_update(
    Passenger *p,
    CruiseShip *ship,
    double delta_time
) {
    if (!p->aboard)
        return;

    /*
     * Passengers spend money
     * while exploring the ship.
     */

    p->money -=
        p->spending_rate *
        delta_time;

    if (p->money < 0)
        p->money = 0;

    /*
     * Satisfaction gradually
     * changes during the voyage.
     */

    p->satisfaction +=
        0.001 *
        delta_time;

    if (p->satisfaction > 1.0)
        p->satisfaction = 1.0;
}

6. Docking
double distance_to_port(
    CruiseShip *ship,
    CruisePort *port
) {
    double dx =
        ship->x - port->x;

    double dy =
        ship->y - port->y;

    return sqrt(
        dx * dx +
        dy * dy
    );
}
bool ship_can_dock(
    CruiseShip *ship,
    CruisePort *port
) {
    return
        distance_to_port(ship, port)
        <= port->docking_radius;
}

typedef enum {

The world determines the current zone:

MaritimeZone get_maritime_zone(
    double x,
    double y
) {
    /*
     * Simplified game-world boundary.
     *
     * Real-world maritime boundaries are
     * much more complicated and should not
     * be treated as exact legal boundaries.
     */

    double distance_from_coast =
        sqrt(
            x * x +
            y * y
        );

    if (distance_from_coast < 10.0)
        return ZONE_PORT;

    if (distance_from_coast < 50.0)
        return ZONE_TERRITORIAL;

    return ZONE_INTERNATIONAL;
}

MaritimeRules get_maritime_rules(
    MaritimeZone zone
) {
    MaritimeRules rules = {0};

    switch (zone) {

        case ZONE_PORT:

            rules.normal_port_rules = true;
            rules.international_navigation = false;
            rules.open_ocean_events = false;

            break;

        case ZONE_TERRITORIAL:

            rules.weather_probability = 0.20;

            break;

        case ZONE_DESTINATION:

            break;
    }

    return rules;
}

Once the ship crosses the line, the game can activate a different event pool.

OceanEvent generate_ocean_event(
    MaritimeRules *rules
) {
    if (!rules->open_ocean_events)
        return EVENT_NONE;

    double r =
        (double)rand() /
        RAND_MAX;

    if (r < 0.02)
        return EVENT_STORM;

    if (r < 0.04)
        return EVENT_MECHANICAL;

    if (r < 0.07)
        return EVENT_OTHER_SHIP;

    if (r < 0.10)
        return EVENT_PARTY;

    return EVENT_NONE;
}

InspectionState inspect_passenger(
    Captain *captain,
    Passenger *passenger
) {
    if (!captain->captain_on_duty)
        return INSPECTION_NORMAL;

    switch (passenger->ticket) {

        case TICKET_VALID:

            return INSPECTION_NORMAL;

        case TICKET_EXPIRED:

        case TICKET_WRONG_CLASS:

    return INSPECTION_NORMAL;

void escalate_ticket_case(
    Captain *captain,
    Passenger *passenger

        passenger->ticket ==
        TICKET_VALID
    ) {
        passenger->state =
            INSPECTION_NORMAL;

        return;

    if (
        passenger->ticket ==

        passenger->ticket ==
        TICKET_WRONG_CLASS

        passenger->state =
            INSPECTION_SUSPICIOUS;

    passenger->state =
        INSPECTION_SECURITY;

Rotor lift
double calculate_lift(
    Helicopter *h,
    HelicopterFlight *f
) {
    double rotor_force =
        h->rotor_power *
        h->rotor_efficiency *
        f->collective;

    /*
     * Scale power into a gameplay-friendly
     * acceleration value.
     */

    return rotor_force /
           h->mass;
}
Gravity
double calculate_vertical_acceleration(
    Helicopter *h,
    HelicopterFlight *f
) {
    double lift =
        calculate_lift(h, f);

    return lift - GRAVITY;
}
Vertical flight
void update_vertical_motion(
    Helicopter *h,
    HelicopterFlight *f,
    double dt
) {
    double acceleration =
        calculate_vertical_acceleration(
            h,
            f
        );

    f->velocity_z +=
        acceleration * dt;

    f->z +=
        f->velocity_z * dt;

    /*
     * Ground collision.
     */

    if (f->z <= 0.0) {

        f->z = 0.0;

        if (f->velocity_z < 0)
            f->velocity_z = 0;

        h->landed = true;
    }
}
Cyclic controls

The cyclic controls the helicopter's movement through pitch and roll.

void update_cyclic(
    HelicopterFlight *f,
    double dt
) {
    f->pitch_rate +=
        f->cyclic_y * 1.5 * dt;

    f->roll_rate +=
        f->cyclic_x * 1.5 * dt;

    f->pitch +=
        f->pitch_rate * dt;

    f->roll +=
        f->roll_rate * dt;

    /*
     * Stability damping.
     */

    f->pitch_rate *= 0.92;
    f->roll_rate *= 0.92;
}
Forward movement
void calculate_forward_motion(
    Helicopter *h,
    HelicopterFlight *f,
    double dt
) {
    double acceleration =
        sin(f->pitch) *
        25.0;

    f->horizontal_speed +=
        acceleration * dt;

    /*
     * Drag.
     */

    f->horizontal_speed *=
        0.995;

    if (
        f->horizontal_speed >
        h->max_speed
    )
        f->horizontal_speed =
            h->max_speed;

    if (
        f->horizontal_speed < 0
    )
        f->horizontal_speed = 0;

    f->velocity_x =
        cos(f->yaw) *
        f->horizontal_speed;

    f->velocity_y =
        sin(f->yaw) *
        f->horizontal_speed;

    f->x +=
        f->velocity_x * dt;

    f->y +=
        f->velocity_y * dt;
}
Yaw
void update_yaw(
    HelicopterFlight *f,
    double dt
) {
    f->yaw_rate +=
        f->rudder *
        1.8 *
        dt;

    f->yaw +=
        f->yaw_rate * dt;

    f->yaw_rate *= 0.90;
}
Complete flight update
void update_helicopter(
    Helicopter *h,
    HelicopterFlight *f,
    double dt
) {
    if (!h->engine_running)
        return;

    update_vertical_motion(
        h,
        f,
        dt
    );

    update_cyclic(
        f,
        dt
    );

    update_yaw(
        f,
        dt
    );

    calculate_forward_motion(
        h,
        f,
        dt
    );

    h->fuel -=
        0.015 *
        f->collective *
        dt;

Dynamic airport congestion

The airport should become busier depending on what's happening in the world.

double calculate_congestion(
    Airport *airport
)
{
    double passenger_ratio =
        (double)airport->current_passengers /
        airport->passenger_capacity;

    double aircraft_ratio =
        (double)airport->current_aircraft /
        MAX_AIRCRAFT;

    double congestion =
        passenger_ratio * 0.7 +
        aircraft_ratio * 0.3;

    return fmin(
        congestion,
        1.0
    );
}

Rare toy economy

The toys have dynamic value.

double calculate_toy_value(
    Toy *toy,
    double collector_demand
) {
    double value =
        toy->value;

    if (toy->rare)
        value *= 2.0;

    value *=
        1.0 + collector_demand;

    return value;
}

A particularly fun final mechanic

Make the Tri-State Area a fictionalised northeastern region of the game world containing the toys' original collector community. Once the player returns, the recovered toys could become permanent objects in a museum, toy shop or collector's mansion.

That also gives you a nice persistent-world effect:

before the quest: empty display cases → during quest: missing-toy investigation → after completion: the recovered toys physically appear in the world.

The airport baggage system, international airline network, Japanese destination zone and collectible economy therefore all become interconnected rather than being isolated quest mechanics.
