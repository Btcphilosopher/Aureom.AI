/* ============================================================
 * RANDOM HELPERS
 * ============================================================ */

static double random_double(void)
{
    return (double)rand() /
           (double)RAND_MAX;
}


static int random_int(
    int min,
    int max
)
{
    return min +
        rand() %
        (max - min + 1);
}

/* ============================================================
 * ANIMAL INITIALISATION
 * ============================================================ */

static void initialise_animal(
    Animal *animal,
    int id,
    Species species,
    WildlifeZone *zone
)
{
    animal->id = id;

    animal->species = species;

    animal->position.x =
        random_double() * zone->width;

    animal->position.y = 0;

    animal->position.z =
        random_double() * zone->length;

    animal->velocity.x = 0;
    animal->velocity.y = 0;
    animal->velocity.z = 0;

    animal->health = 100;

    animal->alive = true;

    animal->state = ROAMING;

    animal->speed = 5.0;

    animal->awareness =
        0.3 + random_double() * 0.6;

    animal->rarity =
        random_double();

    animal->nocturnal = false;

    animal->territorial = false;

    animal->herd_id =
        random_int(0, 10);


    /*
     * Species-specific characteristics.
     */

    switch (species)
    {
        case ALLIGATOR:
            animal->speed = 4;
            animal->territorial = true;
            break;

        case DEER:
            animal->speed = 12;
            break;

        case STAG:
            animal->speed = 13;
            animal->territorial = true;
            break;

        case BOAR:
            animal->speed = 8;
            animal->territorial = true;
            break;

        case BEAR:
            animal->speed = 9;
            animal->territorial = true;
            break;

        case COYOTE:
            animal->speed = 14;
            animal->nocturnal = true;
            break;

        case TIGER:
            animal->speed = 16;
            animal->territorial = true;
            break;

        case JAGUAR:
            animal->speed = 15;
            animal->nocturnal = true;
            break;

        case PANTHER:
            animal->speed = 16;
            animal->nocturnal = true;
            break;

        case WOLF:
            animal->speed = 14;
            break;

        default:
            break;
    }
}


/* ============================================================
 * SPAWN WILDLIFE
 * ============================================================ */

static void spawn_wildlife(
    WildlifeZone *zone,
    int population
)
{
    for (int i = 0;
         i < population &&
         zone->animal_count < MAX_ANIMALS;
         i++)
    {
        int species_index =
            random_int(
                0,
                MAX_TERRAIN_SPECIES - 1
            );

        Species species =
            terrain_species(
                zone->type,
                species_index
            );

        initialise_animal(
            &zone->animals[
                zone->animal_count
            ],
            zone->animal_count,
            species,
            zone
        );

        zone->animal_count++;
    }
}

/* ============================================================
 * ANIMAL AI
 * ============================================================ */

static void update_animal(
    Animal *animal,
    Vec3 player,
    double time_of_day,
    double dt
)
{
    if (!animal->alive)
        return;


    double dx =
        animal->position.x -
        player.x;

    double dz =
        animal->position.z -
        player.z;

    double distance =
        sqrt(
            dx * dx +
            dz * dz
        );


    /*
     * Nocturnal animals become more active at night.
     */

    if (animal->nocturnal)
    {
        if (time_of_day > 20.0 ||
            time_of_day < 6.0)
        {
            animal->state =
                ROAMING;
        }
        else
        {
            animal->state =
                SLEEPING;
        }
    }


    /*
     * Player proximity.
     */

    if (distance <
        80.0 * animal->awareness)
    {
        animal->state =
            FLEEING;
    }


    /*
     * Flee behaviour.
     */

    if (animal->state ==
        FLEEING)
    {
        double magnitude =
            sqrt(
                dx * dx +
                dz * dz
            );

        if (magnitude > 0.01)
        {
            dx /= magnitude;
            dz /= magnitude;
        }

        animal->velocity.x =
            dx * animal->speed;

        animal->velocity.z =
            dz * animal->speed;
    }


    /*
     * Movement.
     */

    animal->position.x +=
        animal->velocity.x * dt;

    animal->position.z +=
        animal->velocity.z * dt;
}


/* ============================================================
 * TRACK GENERATION
 * ============================================================ */

static void generate_tracks(
    Animal *animal
)
{
    if (!animal->alive)
        return;

    printf(
        "[TRACK] %s tracks detected "
        "near %.1f, %.1f\n",
        species_name(animal->species),
        animal->position.x,
        animal->position.z
    );
}


/* ============================================================
 * FICTIONAL GAME HUNTING ACTION
 * ============================================================ */

static bool hunt_action(
    Animal *animal
)
{
    if (!animal->alive)
        return false;

    /*
     * Abstract game action.
     * No real-world weapon modelling.
     */

    animal->alive = false;

    printf(
        "[GAME] Encounter completed: %s #%d\n",
        species_name(animal->species),
        animal->id
    );

    return true;
}


/* ============================================================
 * RARE ENCOUNTER
 * ============================================================ */

static bool is_rare(
    Animal *animal
)
{
    return animal->rarity < 0.03;
}


/* ============================================================
 * ZONE REPORT
 * ============================================================ */

static void zone_report(
    WildlifeZone *zone
)
{
    int living = 0;
    int rare = 0;

    for (int i = 0;
         i < zone->animal_count;
         i++)
    {
        Animal *animal =
            &zone->animals[i];

        if (animal->alive)
        {
            living++;

            if (is_rare(animal))
                rare++;
        }
    }

    printf(
        "\n"
        "============================================\n"
        "%s\n"
        "============================================\n"
        "Terrain:       %s\n"
        "Temperature:   %.1f C\n"
        "Vegetation:    %.2f\n"
        "Water:         %.2f\n"
        "Wildlife:      %d\n"
        "Rare animals:  %d\n"
        "============================================\n",
        zone->name,
        terrain_name(zone->type),
        zone->temperature,
        zone->vegetation,
        zone->water,
        living,
        rare
    );
}


/* ============================================================
 * MAIN DEMONSTRATION
 * ============================================================ */

int main(void)
{
    srand(
        (unsigned int)time(NULL)
    );

    WildlifeZone everglades;
    WildlifeZone desert;
    WildlifeZone jungle;
    WildlifeZone park;

    create_zone(
        &everglades,
        TERRAIN_EVERGLADES,
        "Southern Everglades"
    );

    create_zone(
        &desert,
        TERRAIN_DESERT,
        "Western Desert"
    );

    create_zone(
        &jungle,
        TERRAIN_JUNGLE,
        "Deep Jungle"
    );

    create_zone(
        &park,
        TERRAIN_WILDLIFE_PARK,
        "Malt Whiskey Wildlife Park"
    );


    zone_report(&everglades);
    zone_report(&desert);
    zone_report(&jungle);
    zone_report(&park);


    /*
     * Player simulation.
     */

    Vec3 player = {
        1000,
        0,
        1000
    };


    /*
     * Update wildlife.
     */

    for (int frame = 0;
         frame < 300;
         frame++)
    {
        double time =
            frame * 0.016;

        for (int i = 0;
             i < everglades.animal_count;
             i++)
        {
            update_animal(
                &everglades.animals[i],
                player,
                22.0,
                0.016
            );
        }

        for (int i = 0;
             i < desert.animal_count;
             i++)
        {
            update_animal(
                &desert.animals[i],
                player,
                22.0,
                0.016
            );
        }

        for (int i = 0;
             i < jungle.animal_count;
             i++)
        {
            update_animal(
                &jungle.animals[i],
                player,
                22.0,
                0.016
            );
        }
    }


    /*
     * Find rare encounters.
     */

    printf(
        "\nRARE WILDLIFE ENCOUNTERS\n"
    );

    for (int i = 0;
         i < jungle.animal_count;
         i++)
    {
        if (is_rare(
                &jungle.animals[i]))
        {
            printf(
                "Rare %s detected in jungle.\n",
                species_name(
                    jungle.animals[i].species
                )
            );
        }
    }


    return 0;
}

/* ============================================================
 * INVENTORY
 * ============================================================ */

static bool add_weapon(
    PlayerInventory *inventory,
    Weapon *weapon
)
{
    if (inventory->weapon_count >=
        MAX_INVENTORY)
    {
        printf(
            "Inventory full.\n"
        );

        return false;
    }

    inventory->weapons[
        inventory->weapon_count
    ] = weapon;

    inventory->weapon_count++;

    weapon->owned = true;

    printf(
        "[INVENTORY] Acquired %s\n",
        weapon->name
    );

    return true;
}


/* ============================================================
 * PURCHASE
 * ============================================================ */

static bool purchase_weapon(
    PlayerInventory *inventory,
    Weapon *weapon
)
{
    if (inventory->cash <
        weapon->purchase_price)
    {
        printf(
            "[SHOP] Cannot afford %s\n",
            weapon->name
        );

        return false;
    }

    inventory->cash -=
        weapon->purchase_price;

    return add_weapon(
        inventory,
        weapon
    );
}


/* ============================================================
 * DURABILITY
 * ============================================================ */

static void use_weapon(
    Weapon *weapon
)
{
    if (!weapon->owned)
        return;

    weapon->durability -= 0.05;

    if (weapon->durability < 0)
        weapon->durability = 0;

    if (weapon->durability < 20)
    {
        printf(
            "[WEAPON] %s requires "
            "maintenance.\n",
            weapon->name
        );
    }
}


/* ============================================================
 * WEAPON REPORT
 * ============================================================ */

static void print_weapon(
    Weapon *weapon
)
{
    printf(
        "%-18s | %-10s | %-10s | "
        "DMG %3d | RNG %3.0f | "
        "ACC %.2f | $%.0f\n",

        weapon->name,

        category_name(
            weapon->category
        ),

        rarity_name(
            weapon->rarity
        ),

        weapon->damage,

        weapon->effective_range,

        weapon->accuracy,

        weapon->purchase_price
    );
}


/* ============================================================
 * CATALOGUE REPORT
 * ============================================================ */

static void print_catalogue(
    Weapon weapons[],
    int count
)
{
    printf(
        "\n"
        "============================================================\n"
        "                 WEAPON CATALOGUE\n"
        "============================================================\n"
    );

    for (int i = 0;
         i < count;
         i++)
    {
        print_weapon(
            &weapons[i]
        );
    }

    printf(
        "============================================================\n"
    );
}


/* ============================================================
 * MAIN
 * ============================================================ */

int main(void)
{
    Weapon weapons[MAX_WEAPONS];

    int weapon_count =
        build_weapon_catalogue(

        );


    print_catalogue(

        weapon_count
    );


    /*

    PlayerInventory player;

    memset(
        &player,
        0,
        sizeof(player)
    );

    player.cash = 30000;


    /*

    purchase_weapon(
        &player,

    );

    purchase_weapon(

    );

    purchase_weapon(

    purchase_weapon(

/* ============================================================
 * NPC DAILY ROUTINE
 * ============================================================ */

void update_npc_schedule(
    NPC *npc,
    int hour
)
{
    if (!npc->alive)
        return;


    /*
     * CEO
     */

    if (npc->id == 1)
    {
        if (hour >= 8 && hour < 12)
            npc->current_location =
                LOCATION_CEO_OFFICE;

        else if (hour >= 12 && hour < 14)
            npc->current_location =
                LOCATION_RESTAURANT;

        else if (hour >= 14 && hour < 19)
            npc->current_location =
                LOCATION_BOARDROOM;

        else
            npc->current_location =
                LOCATION_MARINA;
    }


    /*
     * COO
     */

    else if (npc->id == 2)
    {
        if (hour >= 7 && hour < 18)
            npc->current_location =
                LOCATION_FACTORY;

        else
            npc->current_location =
                LOCATION_PENTHOUSE;
    }


    /*
     * CFO
     */

    else if (npc->id == 3)
    {
        if (hour >= 8 && hour < 17)
            npc->current_location =
                LOCATION_BOARDROOM;

        else
            npc->current_location =
                LOCATION_RESTAURANT;
    }


    /*
     * CTO
     */

    else if (npc->id == 4)
    {
        if (hour >= 9 && hour < 22)
            npc->current_location =
                LOCATION_HYDROGEN_LAB;

        else
            npc->current_location =
                LOCATION_BATTERY_LAB;
    }


    /*
     * DESIGNER
     */

    else if (npc->id == 5)
    {
        if (hour >= 10 && hour < 20)
            npc->current_location =
                LOCATION_DESIGN_STUDIO;

        else
            npc->current_location =
                LOCATION_NIGHTCLUB;
    }


    /*
     * RACING EXECUTIVE
     */

    else if (npc->id == 6)
    {
        if (hour >= 8 && hour < 18)
            npc->current_location =
                LOCATION_RACE_TRACK;

        else
            npc->current_location =
                LOCATION_MARINA;
    }


    /*
     * MARKETING EXECUTIVE
     */

    else if (npc->id == 7)
    {
        if (hour >= 9 && hour < 17)
            npc->current_location =
                LOCATION_SUNCOAST_TOWER;

        else if (hour >= 18)
            npc->current_location =
                LOCATION_NIGHTCLUB;

        else
            npc->current_location =
                LOCATION_BEACH;
    }


    /*
     * SECURITY
     */

    else if (npc->id == 8)
    {
        if (hour >= 7 && hour < 23)
            npc->current_location =
                LOCATION_SUNCOAST_TOWER;

        else
            npc->current_location =
                LOCATION_AIRPORT;
    }
}


/* ============================================================
 * NPC MOOD ENGINE
 * ============================================================ */

void update_npc_mood(
    NPC *npc
)
{
    if (npc->stress > 0.85)
    {
        npc->mood =
            MOOD_STRESSED;

        return;
    }

    if (npc->stress > 0.65)
    {
        npc->mood =
            MOOD_FOCUSED;

        return;
    }

    if (npc->personality ==
        PERSONALITY_RACER)
    {
        npc->mood =
            MOOD_EXCITED;

        return;
    }

    if (npc->personality ==
        PERSONALITY_ENGINEER)
    {
        npc->mood =
            MOOD_FOCUSED;

        return;
    }

    if (npc->personality ==
        PERSONALITY_SOCIALITE)
    {
        npc->mood =
            MOOD_HAPPY;

        return;
    }

    npc->mood =
        MOOD_CALM;
}


/* ============================================================
 * PLAYER RELATIONSHIP
 * ============================================================ */

void modify_player_relationship(
    NPC *npc,
    double respect_change,
    double trust_change
)
{
    /*
     * Player is represented as target ID 0.
     */

    int slot = -1;

    for (int i = 0;
         i < npc->relationship_count;
         i++)
    {
        if (npc->relationships[i].target_id == 0)
        {
            slot = i;
            break;
        }
    }


    if (slot == -1)
    {
        if (npc->relationship_count >=
            MAX_RELATIONSHIPS)
            return;

        slot =
            npc->relationship_count++;

        npc->relationships[slot].target_id =
            0;
    }


    npc->relationships[slot].respect +=
        respect_change;

    npc->relationships[slot].trust +=
        trust_change;


    if (npc->relationships[slot].respect >
        1.0)
        npc->relationships[slot].respect = 1.0;

    if (npc->relationships[slot].respect <
        -1.0)
        npc->relationships[slot].respect = -1.0;


    if (npc->relationships[slot].trust >
        1.0)
        npc->relationships[slot].trust = 1.0;

    if (npc->relationships[slot].trust <
        -1.0)
        npc->relationships[slot].trust = -1.0;
}


/* ============================================================
 * NPC REACTION TO PLAYER
 * ============================================================ */

void npc_react_to_player(
    NPC *npc,
    double player_reputation,
    bool player_has_suncoast_vehicle
)
{
    if (!npc->alive)
        return;


    if (npc->personality ==
        PERSONALITY_SOCIALITE)
    {
        if (player_reputation > 0.7)
        {
            modify_player_relationship(
                npc,
                0.05,
                0.03
            );
        }
    }


    if (npc->personality ==
        PERSONALITY_FINANCIER)
    {
        if (player_reputation < 0.2)
        {
            npc->mood =
                MOOD_SUSPICIOUS;
        }
    }


    if (npc->personality ==
        PERSONALITY_RACER)
    {
        if (player_has_suncoast_vehicle)
        {
            npc->mood =
                MOOD_EXCITED;

            modify_player_relationship(
                npc,
                0.04,
                0.01
            );
        }
    }


    if (npc->personality ==
        PERSONALITY_SECURITY)
    {
        if (player_reputation < 0.3)
        {
            npc->mood =
                MOOD_SUSPICIOUS;

            modify_player_relationship(
                npc,
                -0.05,
                -0.08
            );
        }
    }
}

    TexCaldwell *tex,

     *   suspicion is high
     *   demand is high

    double premium =

    if (tex->oil_reserves < 50000000.0)
        premium += 0.15;

    if (tex->suspicion > 0.80)
        premium += 0.10;

    if (tex->reputation > 0.80)
        premium += 0.05;

    return market_price * premium;
}


/* ============================================================
 * OIL -> GOLD TRANSACTION
 * ============================================================ */

bool sell_oil_for_gold(
    TexCaldwell *tex,
    double barrels,
    Gold *gold,
    double oil_market_price,
    double gold_market_price
)
{
    /*
     * Cannot sell more oil than available.
     */

    if (barrels <= 0.0)
        return false;


    if (barrels > tex->oil_reserves)
        return false;


    /*
     * Calculate transaction value.
     */

    double oil_price =
        calculate_oil_price(
            tex,
            oil_market_price
        );


    double total_price =
        barrels *
        oil_price;


    double gold_value =
        calculate_gold_value(
            gold,
            gold_market_price
        );


    /*
     * GOLD ONLY.
     */

    if (gold_value < total_price)
    {
        printf(
            "\nTEX CALDWELL:\n"
            "\"Come back when you've got "
            "the yellow metal.\"\n"
        );

        return false;
    }


    /*
     * Execute transaction.
     */

    tex->oil_reserves -=
        barrels;


    tex->gold_reserves +=
        gold->ounces;


    /*
     * Record transaction.
     */

    if (tex->transaction_count <
        MAX_TRANSACTIONS)
    {
        OilTransaction *transaction =
            &tex->transactions[
                tex->transaction_count++
            ];

        transaction->barrels =
            barrels;

        transaction->gold_paid =
            gold->ounces;

        transaction->completed =
            true;
    }


    printf(
        "\n"
        "================================================\n"
        "CALDWELL ENERGY TRANSACTION\n"
        "================================================\n"
        "Oil purchased: %.0f barrels\n"
        "Gold delivered: %.2f oz\n"
        "Oil price: $%.2f/barrel\n"
        "Settlement: PHYSICAL GOLD\n"
        "================================================\n",
        barrels,
        gold->ounces,
        oil_price
    );


    return true;
}

            MAX_RELATIONSHIPS)

            tex->relationship_count++;


        tex->relationships[index]
            .character_id =
                player_id;
    }


    tex->relationships[index]
        .respect += respect;


    tex->relationships[index]
        .trust += trust;
}

/* ============================================================
 * RANDOM UTILITY
 * ============================================================ */

double random_range(
    double min,
    double max
)
{
    double r =
        (double)rand() /
        (double)RAND_MAX;

    return min +
           r * (max - min);
}


/* ============================================================
 * CLAMP
 * ============================================================ */

double clamp(
    double value,
    double min,
    double max
)
{
    if (value < min)
        return min;

    if (value > max)
        return max;

    return value;
}

/* ============================================================
 * CALCULATE DEMAND
 * ============================================================ */

void calculate_demand(
    OilMarket *market
)
{
    double total = 0.0;

    double industrial = 0.0;

    double transport = 0.0;


    for (int i = 0;
         i < market->region_count;
         i++)
    {
        OilRegion *r =
            &market->regions[i];


        /*
         * Random short-term demand movement.
         */

        double transport_factor =
            1.0 +
            random_range(
                -0.025,
                0.025
            );


        double industrial_factor =
            1.0 +
            random_range(
                -0.020,
                0.020
            );


        r->transport_demand *=
            transport_factor;

        r->industrial_demand *=
            industrial_factor;


        r->total_demand =
            r->industrial_demand +
            r->transport_demand +
            r->electricity_demand;


        total +=
            r->total_demand;

        industrial +=
            r->industrial_demand;

        transport +=
            r->transport_demand;
    }


    market->global_consumption =
        total;

    market->industrial_demand =
        industrial;

    market->transport_demand =
        transport;
}


/* ============================================================
 * PRODUCTION
 * ============================================================ */

void calculate_production(
    OilMarket *market
)
{
    double total = 0.0;


    for (int i = 0;
         i < market->producer_count;
         i++)
    {
        OilProducer *p =
            &market->producers[i];


        if (!p->active)
            continue;


        /*
         * Production varies slightly.
         */

        double output =
            p->daily_production *
            random_range(
                0.97,
                1.03
            );


        /*
         * Producers with very low reserves
         * gradually decline.
         */

        if (p->reserves < 1000000.0)
        {
            output *= 0.80;
        }


        p->reserves -=
            output;


        p->storage +=
            output;


        total +=
            output;
    }


    market->global_production =
        total;
}


/* ============================================================
 * REFINERY DEMAND
 * ============================================================ */

void calculate_refinery_demand(
    OilMarket *market
)
{
    double total = 0.0;


    for (int i = 0;
         i < market->refinery_count;
         i++)
    {
        Refinery *r =
            &market->refineries[i];


        if (!r->active)
            continue;


        double demand =
            r->daily_capacity *
            r->utilization;


        r->demand =
            demand;


        total +=
            demand;
    }


    market->refinery_demand =
        total;
}


/* ============================================================
 * INVENTORY
 * ============================================================ */

void update_inventory(
    OilMarket *market
)
{
    double net_flow =
        market->global_production
        -
        market->global_consumption;


    market->global_inventory +=
        net_flow;


    if (market->global_inventory <
        0.0)
    {
        market->global_inventory =
            0.0;
    }
}


/* ============================================================
 * MARKET SENTIMENT
 * ============================================================ */

void calculate_sentiment(
    OilMarket *market
)
{
    double supply =
        market->global_production;

    double demand =
        market->global_consumption;


    market->sentiment.supply_pressure =
        supply / fmax(
            demand,
            1.0
        );


    market->sentiment.demand_pressure =
        demand / fmax(
            supply,
            1.0
        );


    /*
     * Inventory pressure.
     */

    double inventory_ratio =
        market->global_inventory /
        fmax(
            demand * 30.0,
            1.0
        );


    market->sentiment.inventory_pressure =
        clamp(
            inventory_ratio,
            0.0,
            3.0
        );


    /*
     * Random market uncertainty.
     */

    market->sentiment.geopolitical_risk =
        random_range(
            0.0,
            0.20
        );


    market->sentiment.weather_risk =
        random_range(
            0.0,
            0.10
        );


    market->sentiment.speculation =
        random_range(
            -0.10,
            0.10
        );


    market->sentiment.confidence =
        clamp(
            0.75 -
            market->sentiment.geopolitical_risk
            -
            market->sentiment.weather_risk,
            0.0,
            1.0
        );
}


/* ============================================================
 * PRICE ENGINE
 * ============================================================ */

void update_oil_price(
    OilMarket *market
)
{
    double price =
        market->price.spot_price;


    /*
     * Supply / demand balance.
     */

    double balance =
        market->global_consumption -
        market->global_production;


    double demand_effect =
        balance /
        fmax(
            market->global_consumption,
            1.0
        );


    /*
     * Inventory effect.
     *
     * Low inventory = higher prices.
     */

    double inventory_target =
        market->global_consumption *
        30.0;


    double inventory_ratio =
        market->global_inventory /
        fmax(
            inventory_target,
            1.0
        );


    double inventory_effect =
        0.50 -
        inventory_ratio;


    /*
     * Risk premium.
     */

    double risk =
        market->sentiment
            .geopolitical_risk
        +
        market->sentiment
            .weather_risk;


    /*
     * Speculative component.
     */

    double speculation =
        market->sentiment
            .speculation;


    /*
     * Combined price movement.
     */

    double change =
        demand_effect * 0.20
        +
        inventory_effect * 0.10
        +
        risk * 0.10
        +
        speculation * 0.05;


    /*
     * Add small market noise.
     */

    change +=
        random_range(
            -0.015,
            0.015
        );


    price *=
        1.0 + change;


    price =
        clamp(
            price,
            MIN_OIL_PRICE,
            MAX_OIL_PRICE
        );


    market->price.previous_price =
        market->price.spot_price;

    market->price.spot_price =
        price;


    if (price >
        market->price.daily_high)
    {
        market->price.daily_high =
            price;
    }


    if (price <
        market->price.daily_low)
    {
        market->price.daily_low =
            price;
    }
}


/* ============================================================
 * STRATEGIC RESERVE
 * ============================================================ */

void manage_strategic_reserve(
    OilMarket *market
)
{
    double demand =
        market->global_consumption;


    double reserve_days =
        market->strategic_reserve /
        fmax(
            demand,
            1.0
        );


    /*
     * Build reserve when prices are low.
     */

    if (reserve_days < 60.0 &&
        market->price.spot_price < 70.0)
    {
        double purchase =
            demand * 0.10;

        market->strategic_reserve +=
            purchase;

        market->global_inventory -=
            purchase;
    }


    /*
     * Release reserve during shortage.
     */

    if (reserve_days > 30.0 &&
        market->price.spot_price > 150.0)
    {
        double release =
            demand * 0.05;

        market->strategic_reserve -=
            release;

        market->global_inventory +=
            release;
    }
}


/* ============================================================
 * MARKET SHOCK
 * ============================================================ */

void apply_market_shock(
    OilMarket *market,
    double production_multiplier,
    double demand_multiplier
)
{
    for (int i = 0;
         i < market->producer_count;
         i++)
    {
        market->producers[i]
            .daily_production *=
                production_multiplier;
    }


    for (int i = 0;
         i < market->region_count;
         i++)
    {
        market->regions[i]
            .transport_demand *=
                demand_multiplier;

        market->regions[i]
            .industrial_demand *=
                demand_multiplier;
    }
}


/* ============================================================
 * DAILY SIMULATION
 * ============================================================ */

void simulate_day(
    OilMarket *market
)
{
    market->day++;


    market->price.opening_price =
        market->price.spot_price;

    market->price.daily_high =
        market->price.spot_price;

    market->price.daily_low =
        market->price.spot_price;

        market

    calculate_production(
        market
    );


    calculate_refinery_demand(
        market
    );


    update_inventory(
        market
    );


    calculate_sentiment(
        market
    );


    manage_strategic_reserve(
        market
    );


    update_oil_price(
        market
    );
}

#define MAX_ORDER_SIZE    100000.0

#define SECURITY_NORMAL   0
#define SECURITY_ELEVATED 1

{

    double market_value;

    bool assayed;

    bool vaulted;

    bool locked;

} GoldBar;

 * MEMBER
 * ============================================================ */

typedef struct
{
    int id;

    char name[64];

    char organisation[96];

    double cash_balance;

    double gold_balance;

    int security_clearance;

    bool active;

} ExchangeMember;

 * ORDER
 * ============================================================ */

typedef enum
{
    ORDER_BUY,
    ORDER_SELL

typedef struct
{
    int id;

    int member_id;

    OrderType type;

    double ounces;

    double price;

 * ============================================================ */

typedef struct

 * AUDIT
 * ============================================================ */

typedef struct
{

 * SECURITY

typedef struct
{

/* ============================================================
 * AUDIT LOG
 * ============================================================ */

void audit_log(
    GoldExchange *exchange,
    int member_id,
    const char *action,
    const char *result
)
{
    if (exchange->audit_count >= MAX_AUDIT)
        return;

    AuditEntry *entry =
        &exchange->audit[
            exchange->audit_count++
        ];

    entry->timestamp =
        time(NULL);

    entry->member_id =
        member_id;

    snprintf(
        entry->action,
        sizeof(entry->action),
        "%s",
        action
    );

    snprintf(
        entry->result,
        sizeof(entry->result),
        "%s",
        result
    );
}


/* ============================================================
 * FIND MEMBER
 * ============================================================ */

ExchangeMember *find_member(
    GoldExchange *exchange,
    int id
)
{
    for (int i = 0;
         i < exchange->member_count;
         i++)
    {
        if (exchange->members[i].id == id)
            return &exchange->members[i];
    }

    return NULL;
}

        );
    }

    else
    {
        printf(
            "\n[SECURITY] STATUS: LOCKDOWN\n"

    }
}

/* ============================================================
 * GOLD ASSAY
 * ============================================================ */

bool assay_gold(
    GoldBar *bar
)
{
    if (bar->weight_oz <= 0.0)
        return false;


    if (bar->purity <
        GOLD_MIN_PURITY)
        return false;


    bar->assayed =
        true;


    return true;
}


/* ============================================================
 * DEPOSIT GOLD
 * ============================================================ */

bool deposit_gold(
    GoldExchange *exchange,
    int member_id,
    double ounces,
    double purity
)
{
    if (!exchange->deposits_enabled)
        return false;


    if (exchange->security_state ==
        SECURITY_LOCKDOWN)
        return false;


    if (ounces <= 0.0)
        return false;


    if (purity < GOLD_MIN_PURITY)
        return false;


    if (exchange->bar_count >=
        MAX_BARS)
        return false;


    ExchangeMember *member =
        find_member(
            exchange,
            member_id
        );


    if (!member ||
        !member->active)
        return false;


    GoldBar *bar =
        &exchange->bars[
            exchange->bar_count++
        ];


    bar->id =
        exchange->bar_count;


    bar->weight_oz =
        ounces;


    bar->purity =
        purity;


    bar->market_value =
        ounces *
        purity *
        exchange->gold_spot_price;


    bar->assayed =
        assay_gold(bar);


    bar->vaulted =
        true;


    bar->locked =
        false;


    member->gold_balance +=
        ounces;


    exchange->total_gold_vaulted +=
        ounces;


    audit_log(
        exchange,
        member_id,
        "GOLD_DEPOSIT",
        "SETTLED"
    );


    return true;
}


/* ============================================================
 * WITHDRAW GOLD
 * ============================================================ */

bool withdraw_gold(
    GoldExchange *exchange,
    int member_id,
    double ounces
)
{
    if (!exchange->withdrawals_enabled)
        return false;


    if (exchange->security_state ==
        SECURITY_LOCKDOWN)
        return false;


    ExchangeMember *member =
        find_member(
            exchange,
            member_id
        );


    if (!member ||
        !member->active)
        return false;


    if (member->gold_balance <
        ounces)
        return false;


    member->gold_balance -=
        ounces;


    exchange->total_gold_vaulted -=
        ounces;


    audit_log(
        exchange,
        member_id,
        "GOLD_WITHDRAWAL",
        "APPROVED"
    );


    return true;
}


/* ============================================================
 * BUY ORDER
 * ============================================================ */

int place_buy_order(
    GoldExchange *exchange,
    int member_id,
    double ounces,
    double price
)
{
    if (!exchange->market_open)
        return -1;


    if (exchange->security_state ==
        SECURITY_LOCKDOWN)
        return -1;


    if (ounces <= 0.0 ||
        ounces > MAX_ORDER_SIZE)
        return -1;


    if (price <= 0.0)
        return -1;


    ExchangeMember *member =
        find_member(
            exchange,
            member_id
        );


    if (!member ||
        !member->active)
        return -1;


    double required_cash =
        ounces * price;


    if (member->cash_balance <
        required_cash)
        return -1;


    if (exchange->order_count >=
        MAX_ORDERS)
        return -1;


    GoldOrder *order =
        &exchange->orders[
            exchange->order_count
        ];


    order->id =
        exchange->order_count;


    order->member_id =
        member_id;


    order->type =
        ORDER_BUY;


    order->ounces =
        ounces;


    order->price =
        price;


    order->remaining =
        ounces;


    order->active =
        true;


    exchange->order_count++;


    audit_log(
        exchange,
        member_id,
        "BUY_ORDER",
        "ACCEPTED"
    );


    return order->id;
}

/* ============================================================
 * SETTLE TRADE
 * ============================================================ */

void settle_trade(
    GoldExchange *exchange,
    GoldOrder *buy,
    GoldOrder *sell,
    double ounces,
    double price
)
{
    ExchangeMember *buyer =
        find_member(
            exchange,
            buy->member_id
        );


    ExchangeMember *seller =
        find_member(
            exchange,
            sell->member_id
        );


    if (!buyer || !seller)
        return;


    double value =
        ounces * price;


    /*
     * Cash settlement.
     */

    buyer->cash_balance -=
        value;


    seller->cash_balance +=
        value;


    /*
     * Gold settlement.
     */

    buyer->gold_balance +=
        ounces;


    seller->gold_balance -=
        ounces;


    /*
     * Record trade.
     */

    if (exchange->trade_count <
        MAX_TRADES)
    {
        GoldTrade *trade =
            &exchange->trades[
                exchange->trade_count++
            ];


        trade->id =
            exchange->trade_count;


        trade->buyer_id =
            buyer->id;


        trade->seller_id =
            seller->id;


        trade->ounces =
            ounces;


        trade->price =
            price;


        trade->value =
            value;


        trade->timestamp =
            time(NULL);
    }


    buy->remaining -=
        ounces;


    sell->remaining -=
        ounces;


    if (buy->remaining <= 0.0001)
        buy->active = false;


    if (sell->remaining <= 0.0001)
        sell->active = false;


    exchange->gold_spot_price =
        price;


    printf(
        "\n"
        "[TRADE EXECUTED]\n"
        "Gold: %.2f oz\n"
        "Price: $%.2f\n"
        "Value: $%.2f\n",
        ounces,
        price,
        value
    );
}


/* ============================================================
 * ORDER MATCHING
 * ============================================================ */

void match_orders(
    GoldExchange *exchange
)
{
    for (int i = 0;
         i < exchange->order_count;
         i++)
    {
        GoldOrder *buy =
            &exchange->orders[i];


        if (!buy->active ||
            buy->type != ORDER_BUY)
            continue;


        for (int j = 0;
             j < exchange->order_count;
             j++)
        {
            GoldOrder *sell =
                &exchange->orders[j];


            if (!sell->active ||
                sell->type != ORDER_SELL)
                continue;


            if (buy->price <
                sell->price)
                continue;


            double volume =
                fmin(
                    buy->remaining,
                    sell->remaining
                );


            settle_trade(
                exchange,
                buy,
                sell,
                volume,
                sell->price
            );


            if (!buy->active)
                break;
        }
    }
}


/* ============================================================
 * SECURITY MONITOR
 * ============================================================ */

void security_scan(
    GoldExchange *exchange
)
{
    int suspicious_orders = 0;


    for (int i = 0;
         i < exchange->order_count;
         i++)
    {
        GoldOrder *order =
            &exchange->orders[i];


        if (!order->active)
            continue;


        if (order->ounces >
            MAX_ORDER_SIZE * 0.80)

 * ============================================================
 * HIDDEN WEALTH DIFFICULTY SYSTEM
 * ============================================================
 *
 * Invisible gameplay modifier.
 *
 * The player never sees:
 *
 *     wealth_difficulty
 *     wealth_pressure
 *     preservation_pressure
 *
 * The system quietly adjusts the difficulty of maintaining
 * large amounts of money.
 * ============================================================
 */

#include <math.h>
#include <stdbool.h>

typedef struct
{
    double cash;
    double assets;

    float wealth_difficulty;
    float financial_pressure;

} HiddenWealthState;


/*
 * Convert wealth into a hidden difficulty value.
 *
 * Logarithmic scaling prevents billionaires from simply
 * reaching an absurd numerical difficulty value.
 */

static float calculate_wealth_difficulty(
    double total_wealth
)
{
    if (total_wealth <= 10000.0)
        return 0.0f;

    double x =
        log10(total_wealth / 10000.0);

    float difficulty =
        (float)(x / 8.0);

    if (difficulty < 0.0f)
        difficulty = 0.0f;

    if (difficulty > 1.0f)
        difficulty = 1.0f;

    return difficulty;
}
Hidden update
void update_hidden_wealth_system(
    HiddenWealthState *state
)
{
    double total =
        state->cash +
        state->assets;

    state->wealth_difficulty =
        calculate_wealth_difficulty(total);

    /*
     * Smooth pressure prevents sudden difficulty jumps.
     */

    state->financial_pressure =
        state->wealth_difficulty *
        0.85f;

typedef struct
{
    float current;
    float target;

} DifficultyController;


void smooth_difficulty(
    DifficultyController *d,
    float target,
    float dt
)
{
    const float rate = 0.15f;

    d->target = target;

    d->current +=
        (d->target - d->current)
        * rate
        * dt;

    if (d->current < 0.0f)
        d->current = 0.0f;

    if (d->current > 1.0f)
        d->current = 1.0f;
}

double calculate_price(
    const ContrabandMarket *market
)
{
    if (market->supply <= 0.01)
        return market->base_price * 4.0;

    double ratio =
        market->demand /
        market->supply;

    double price =
        market->base_price *
        (0.5 + ratio);

    return price;
}

        m->demand /
        fmax(m->supply, 0.01);

    MarketEvent event

    switch (event)
    {

            market->supply *= 0.55;
            market->volatility += 0.15f;
            break;

        case MARKET_SHORTAGE:
            market->supply *= 0.35;

            break;

Inflation engine
void update_money_economy(
    MoneyEconomy *economy
)
{
    economy->total_currency =
        economy->legitimate_currency +
        economy->fraudulent_currency;

    if (economy->legitimate_currency <= 0.0)
        return;

    double expansion =
        economy->fraudulent_currency /
        economy->legitimate_currency;

    economy->inflation_rate =
        expansion * 0.12;

    economy->price_index =
        1.0 +
        economy->inflation_rate;

    if (expansion < 0.05)
        economy->state = MONEY_NORMAL;

    else if (expansion < 0.15)
        economy->state = MONEY_EXPANDING;

    else if (expansion < 0.30)
        economy->state = MONEY_INFLATION;

    else if (expansion < 0.60)
        economy->state = MONEY_CRISIS;

    else
        economy->state = MONEY_COLLAPSE;
}
Malt Whiskey production
void update_malt_whiskey(
    MaltWhiskeyMoneyPlant *plant,
    MoneyEconomy *economy,
    float dt
)
{
    if (!plant->printer_active)
        return;

    double produced =
        plant->production_rate * dt;

    plant->money_stock += produced;

    economy->fraudulent_currency +=
        produced;

    /*
     * More production means more economic instability.
     */

    plant->secrecy -=
        produced * 0.00001f;

    if (plant->secrecy < 0.0f)
        plant->secrecy = 0.0f;
}

Dynamic pricing
double calculate_food_price(
    FoodProduct *product
)
{
    if (product->supply <= 0.01)
        return product->base_price * 5.0;

    double ratio =
        product->demand /
        product->supply;

    double price =
        product->base_price *
        (0.5 + ratio);

    /*
     * Fresh products command
     * higher prices.
     */

    price *=
        (0.5 + product->freshness * 0.5);

    return price;
}

Player selling
typedef struct
{
    double quantity;
    FoodProduct product;

} PlayerFoodInventory;


double sell_food(
    FoodProduct *product,
    double quantity
)
{
    if (quantity <= 0.0)
        return 0.0;

    if (quantity > product->supply)
        quantity = product->supply;

    double revenue =
        quantity *
        product->market_price;

    product->supply += quantity;

    return revenue;
}
