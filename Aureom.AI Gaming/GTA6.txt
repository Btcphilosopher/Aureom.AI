Demand-pull economy

The key mechanic should not simply be:

city needs food → spawn food

Instead:

population
     ↓
consumption
     ↓
inventory
     ↓
shortage
     ↓
price increase
     ↓
profit opportunity
     ↓
production response
     ↓
rail freight demand
economy.c
#include <math.h>
#include "world.h"

static float clamp(float value, float min, float max)
{
    if (value < min) return min;
    if (value > max) return max;
    return value;
}

void update_city_demand(City *city, float dt)
{
    for (int g = 0; g < GOODS_COUNT; g++) {

        float base_demand =
            city->consumption[g] *
            city->population;

        /*
         * Demand rises when inventory falls.
         *
         * This creates the economic "pull".
         */
        float inventory_ratio =
            city->inventory[g] /
            fmaxf(base_demand * 7.0f, 1.0f);

        float shortage =
            clamp(1.0f - inventory_ratio, 0.0f, 1.0f);

        city->demand[g] =
            base_demand *
            (1.0f + shortage * 2.0f);

        /*
         * Simple scarcity pricing.
         */
        float target_price =
            10.0f *
            (1.0f + shortage * 4.0f);

        /*
         * Prices move gradually rather than
         * instantly jumping.
         */
        city->price[g] +=
            (target_price - city->price[g])
            * 0.05f *
            dt;
    }
}

Railway system

The railway should be a physical economic network, rather than merely a transportation animation.

railway.c
#include <math.h>
#include "world.h"

static float distance(Vec2 a, Vec2 b)
{
    float dx = b.x - a.x;
    float dy = b.y - a.y;

    return sqrtf(dx * dx + dy * dy);
}

void update_train(
    World *world,
    Train *train,
    float dt
)
{
    if (!train->active)
        return;

    Station *from =
        &world->stations[train->current_station];

    Station *to =
        &world->stations[train->destination_station];

    float route_distance =
        distance(from->position, to->position);

    if (route_distance <= 0.01f)
        return;

    /*
     * Convert physical velocity into
     * normalized route progress.
     */
    float movement =
        (train->speed * dt) /
        route_distance;

    train->progress += movement;

    if (train->progress >= 1.0f) {

        train->progress = 0.0f;

        /*
         * Arrived.
         */
        train->current_station =
            train->destination_station;

        /*
         * Example destination-selection logic.
         * A real implementation would use
         * the economic routing engine.
         */
        train->destination_station =
            (train->destination_station + 1)
            % world->station_count;
    }

    train->fuel -=
        train->operating_cost * dt;

    if (train->fuel <= 0.0f) {
        train->active = 0;
    }
}
Economic freight routing

This is where it gets much more interesting.

Instead of manually telling trains what to carry, calculate:

profit =
destination_price
-
origin_price
-
transport_cost
-
handling_cost
-
fuel_cost
-
risk
demand.c
#include "world.h"

typedef struct {
    int origin;
    int destination;

    GoodType good;

    float quantity;
    float expected_profit;

} FreightOpportunity;

FreightOpportunity find_best_freight(
    World *world
)
{
    FreightOpportunity best = {
        -1, -1, GOODS_FOOD, 0.0f, -999999.0f
    };

    for (int a = 0; a < world->station_count; a++) {

        for (int b = 0; b < world->station_count; b++) {

            if (a == b)
                continue;

            for (int g = 0; g < GOODS_COUNT; g++) {

                /*
                 * For the prototype we map stations
                 * to cities by index.
                 */
                if (a >= world->city_count ||
                    b >= world->city_count)
                    continue;

                City *origin =
                    &world->cities[a];

                City *destination =
                    &world->cities[b];

                float price_origin =
                    origin->price[g];

                float price_destination =
                    destination->price[g];

                float price_difference =
                    price_destination -
                    price_origin;

                float transport_cost =
                    2.0f;

                float profit =
                    price_difference -
                    transport_cost;

                if (profit >
                    best.expected_profit) {

                    best.origin = a;
                    best.destination = b;
                    best.good = g;

                    best.quantity = 100.0f;

                    best.expected_profit =
                        profit;
                }
            }
        }
    }

    return best;
}

This gives you a basic AI freight market.

The live simulation loop
simulation.c
#include "world.h"

void update_city_demand(City *, float);
void update_train(World *, Train *, float);

void simulation_tick(
    World *world,
    float dt
)
{
    /*
     * 1. Economic demand
     */
    for (int i = 0;
         i < world->city_count;
         i++) {

        update_city_demand(
            &world->cities[i],
            dt
        );
    }

    /*
     * 2. Industrial production
     */
    for (int i = 0;
         i < world->industry_count;
         i++) {

        Industry *industry =
            &world->industries[i];

        industry->inventory +=
            industry->production_rate * dt;
    }

    /*
     * 3. Railway movement
     */
    for (int i = 0;
         i < world->train_count;
         i++) {

        update_train(
            world,
            &world->trains[i],
            dt
        );
    }

    /*
     * 4. Advance world clock.
     */
    world->simulation_time += dt;
}

And the game can run it independently of rendering:

#include <stdio.h>
#include "world.h"

void simulation_tick(World *, float);

int main(void)
{
    World world = {0};

    /*
     * Populate world here.
     */

    const float dt = 1.0f / 20.0f;

    for (;;) {

        simulation_tick(
            &world,
            dt
        );

        /*
         * Game engine consumes
         * the resulting world state.
         */

        printf(
            "Simulation time: %.2f\n",
            world.simulation_time
        );
    }

    return 0;
}
The bigger architecture I'd use

For the full game, I'd separate simulation from the game engine:

                 OPEN-WORLD GAME
                       │
       ┌───────────────┼────────────────┐
       │               │                │
   PLAYER AI       VEHICLES          NPCs
       │               │                │
       └───────────────┼────────────────┘
                       │
                 GAME STATE API
                       │
                       ▼
              ┌─────────────────┐
              │    WORLD SIM    │
              │      C CORE     │
              └─────────────────┘
                       │
       ┌───────────────┼─────────────────┐
       ▼               ▼                 ▼
    ECONOMY        RAILWAY           INDUSTRY
       │               │                 │
       ▼               ▼                 ▼

       │               │                 │
       └───────────────┼─────────────────┘
                       ▼
                    PRICES
                       │
                       └──────→ DEMAND

The particularly powerful feature would be economic emergence.

A player could, for example, cause a huge increase in construction activity in one city. That creates:

construction boom → steel shortage → steel prices ↑ → coal demand ↑ → coal freight ↑ → railway utilisation ↑ → locomotive demand ↑ → industrial employment ↑ → population ↑ → food demand ↑ → food freight ↑.

Meanwhile, another city could experience a recession and see the reverse.

The railway therefore becomes part of the simulation's economic nervous system, rather than simply scenery or a transport minigame.

For a GTA-style open-world game, I'd keep the C simulation game-agnostic and expose it through a compact API so Unreal/Unity/custom C/C++ rendering code can query things like:

get_city_price(city_id, GOODS_STEEL);
get_city_demand(city_id, GOODS_FOOD);
get_station_cargo(station_id, GOODS_COAL);
get_train_position(train_id);
get_freight_opportunity();
get_city_economic_state(city_id);

That gives you a genuinely interesting living open-world economy + railway simulation underneath the visual game world.

/* ============================================================
   PURCHASE DINING ITEM
   ============================================================ */

double purchase_menu_item(
    LuxuryTrain *train,
    int passenger_id,
    int menu_id
)
{
    if (passenger_id < 0 ||
        passenger_id >= train->passenger_count)
        return 0.0;

    if (menu_id < 0 ||
        menu_id >= train->menu_count)
        return 0.0;

    MenuItem *item =
        &train->menu[menu_id];

    if (item->inventory <= 0)
        return 0.0;

    Passenger *passenger =
        &train->passengers[passenger_id];

    /*
     * Premium passengers have higher
     * propensity to purchase premium items.
     */
    if (item->price >
        passenger->willingness_to_pay)
        return 0.0;

    item->inventory--;

    passenger->spending +=
        item->price;

    passenger->satisfaction +=
        1.0;

    passenger->satisfaction =
        clamp(
            passenger->satisfaction,
            0.0,
            100.0
        );

    if (
        item->category == ITEM_WINE ||
        item->category == ITEM_SPIRIT ||
        item->category == ITEM_COCKTAIL
    )
    {
        train->beverage_revenue +=
            item->price;
    }
    else
    {
        train->dining_revenue +=
            item->price;
    }

    return item->price;
}


/* ============================================================
   SERVICE QUALITY
   ============================================================ */

void update_service_quality(
    LuxuryTrain *train,
    double dt
)
{
    double total_skill = 0.0;

    int active_staff = 0;

    for (int i = 0;
         i < train->staff_count;
         i++)
    {
        Staff *s =
            &train->staff[i];

        if (!s->active)
            continue;

        total_skill +=
            s->service_skill;

        active_staff++;

        /*
         * Staff fatigue accumulates
         * during a long journey.
         */
        s->fatigue +=
            0.01 * dt;

        s->fatigue =
            clamp(
                s->fatigue,
                0.0,
                50.0
            );
    }

    if (active_staff == 0)
        return;

    double average_skill =
        total_skill /
        active_staff;

    /*
     * Service quality combines:
     *
     * staff skill
     * cleanliness
     * car service level
     * staffing
     */

    double car_quality = 0.0;

    int cars = 0;

    for (int i = 0;
         i < train->car_count;
         i++)
    {
        car_quality +=
            train->cars[i].service_level;

        car_quality +=
            train->cars[i].cleanliness;

        cars++;
    }

    if (cars > 0)
        car_quality /=
            (cars * 2.0);

    double quality =
        average_skill * 0.40 +
        car_quality * 0.60;

    train->satisfaction +=
        (quality - train->satisfaction)
        * 0.01
        * dt;

    train->satisfaction =
        clamp(
            train->satisfaction,
            0.0,
            100.0
        );
}


/* ============================================================
   OPERATING COST
   ============================================================ */

void update_operating_cost(
    LuxuryTrain *train,
    double dt
)
{
    for (int i = 0;
         i < train->car_count;
         i++)
    {
        train->operating_cost +=
            train->cars[i].operating_cost
            * dt;
    }

    for (int i = 0;
         i < train->staff_count;
         i++)
    {
        Staff *s =
            &train->staff[i];

        if (s->active)
        {
            train->operating_cost +=
                s->hourly_cost * dt;
        }
    }
}


/* ============================================================
   PREMIUM TICKET REVENUE
   ============================================================ */

double ticket_price(
    PassengerClass class,
    double distance_miles
)
{
    double base;

    switch (class)
    {
        case PASSENGER_FIRST_CLASS:
            base = 1.75;
            break;

        case PASSENGER_SUITE:
            base = 4.50;
            break;

        default:
            base = 1.00;
            break;
    }

    return base *
           distance_miles;
}

#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>
#include <stdbool.h>
#include <string.h>
#include <time.h>

#define MAX_ITEMS       256
#define MAX_CRYPTO      32
#define MAX_NAME        64
#define MAX_LOG_ENTRIES 128

typedef enum {
    ITEM_CURRENCY,
    ITEM_WEAPON,
    ITEM_EQUIPMENT,
    ITEM_VEHICLE,
    ITEM_COLLECTIBLE,
    ITEM_DOCUMENT,
    ITEM_MISSION,
    ITEM_CONTRABAND
} ItemType;

typedef struct {
    uint64_t id;
    ItemType type;

    char name[MAX_NAME];

    uint32_t quantity;
    uint64_t value;

    bool insured;
    bool mission_locked;
} CryptItem;

typedef struct {
    char address[MAX_CRYPTO];
    uint64_t balance;
} CryptoWallet;

typedef struct {
    uint64_t timestamp;
    char event[MAX_NAME];
} CryptLog;

typedef struct {

    /* Player-owned assets */
    CryptItem items[MAX_ITEMS];
    size_t item_count;

    /* In-game digital currency */
    CryptoWallet wallets[4];
    size_t wallet_count;

    /* Activity history */
    CryptLog logs[MAX_LOG_ENTRIES];
    size_t log_count;

    /* Vault properties */
    uint32_t security_level;
    uint64_t storage_capacity;

} DigitalCrypt;


/* ============================================================
 * INITIALISATION
 * ============================================================ */

void crypt_init(DigitalCrypt *crypt)
{
    if (!crypt)
        return;

    memset(crypt, 0, sizeof(DigitalCrypt));

    crypt->security_level = 1;
    crypt->storage_capacity = 100;

    printf("[CYBERCRYPT] Digital vault initialised.\n");
}


/* ============================================================
 * LOGGING
 * ============================================================ */

static void crypt_log(DigitalCrypt *crypt, const char *event)
{
    if (!crypt || !event)
        return;

    if (crypt->log_count >= MAX_LOG_ENTRIES)
        return;

    CryptLog *log = &crypt->logs[crypt->log_count++];

    log->timestamp = (uint64_t)time(NULL);

    snprintf(
        log->event,
        sizeof(log->event),
        "%s",
        event
    );
}


/* ============================================================
 * ITEM STORAGE
 * ============================================================ */

bool crypt_store_item(
    DigitalCrypt *crypt,
    uint64_t id,
    ItemType type,
    const char *name,
    uint32_t quantity,
    uint64_t value
)
{
    if (!crypt || !name)
        return false;

    if (crypt->item_count >= MAX_ITEMS)
        return false;

    CryptItem *item = &crypt->items[crypt->item_count];

    item->id = id;
    item->type = type;

    snprintf(
        item->name,
        sizeof(item->name),
        "%s",
        name
    );

    item->quantity = quantity;
    item->value = value;

    item->insured = false;
    item->mission_locked = false;

    crypt->item_count++;

    char log_message[MAX_NAME];

    snprintf(
        log_message,
        sizeof(log_message),
        "Stored asset: %s",
        name
    );

    crypt_log(crypt, log_message);

    return true;
}


/* ============================================================
 * ITEM RETRIEVAL
 * ============================================================ */

bool crypt_withdraw_item(
    DigitalCrypt *crypt,
    uint64_t id
)
{
    if (!crypt)
        return false;

    for (size_t i = 0; i < crypt->item_count; ++i) {

        if (crypt->items[i].id == id) {

            if (crypt->items[i].mission_locked)
                return false;

            char log_message[MAX_NAME];

            snprintf(
                log_message,
                sizeof(log_message),
                "Retrieved asset: %s",
                crypt->items[i].name
            );

            crypt_log(crypt, log_message);

            /*
             * Compact array after removal.
             */

            for (size_t j = i;
                 j < crypt->item_count - 1;
                 ++j)
            {
                crypt->items[j] = crypt->items[j + 1];
            }

            crypt->item_count--;

            return true;
        }
    }

    return false;
}


/* ============================================================
 * IN-GAME DIGITAL CURRENCY
 * ============================================================ */

bool crypt_deposit_currency(
    DigitalCrypt *crypt,
    uint64_t amount
)
{
    if (!crypt || crypt->wallet_count == 0)
        return false;

    crypt->wallets[0].balance += amount;

    crypt_log(
        crypt,
        "Currency deposited into digital vault"
    );

    return true;
}


bool crypt_withdraw_currency(
    DigitalCrypt *crypt,
    uint64_t amount
)
{
    if (!crypt || crypt->wallet_count == 0)
        return false;

    if (crypt->wallets[0].balance < amount)
        return false;

    crypt->wallets[0].balance -= amount;

    crypt_log(
        crypt,
        "Currency withdrawn from digital vault"
    );

    return true;
}


/* ============================================================
 * WALLET CREATION
 * ============================================================ */

bool crypt_create_wallet(
    DigitalCrypt *crypt,
    const char *address
)
{
    if (!crypt || !address)
        return false;

    if (crypt->wallet_count >= 4)
        return false;

    CryptoWallet *wallet =
        &crypt->wallets[crypt->wallet_count];

    snprintf(
        wallet->address,
        sizeof(wallet->address),
        "%s",
        address
    );

    wallet->balance = 0;

    crypt->wallet_count++;

    crypt_log(
        crypt,
        "New in-game digital wallet created"
    );

    return true;
}


/* ============================================================
 * VAULT SECURITY
 * ============================================================ */

void crypt_upgrade_security(
    DigitalCrypt *crypt
)
{
    if (!crypt)
        return;

    if (crypt->security_level < 10)
        crypt->security_level++;

    crypt->storage_capacity += 25;

    crypt_log(
        crypt,
        "Crypt security upgraded"
    );
}

    if (!crypt)

    );

    printf(

        crypt->item_count
    );

    printf("\nASSETS\n");

    for (size_t i = 0; i < crypt->item_count; ++i) {

        printf(

/* ============================================================
 * CRYPT VALUATION
 * ============================================================ */

uint64_t crypt_total_value(
    const DigitalCrypt *crypt
)
{
    if (!crypt)
        return 0;

    uint64_t total = 0;

    for (size_t i = 0; i < crypt->item_count; ++i) {

        total +=
            crypt->items[i].value *
            crypt->items[i].quantity;
    }

    for (size_t i = 0; i < crypt->wallet_count; ++i)
        total += crypt->wallets[i].balance;

    return total;
}

double calculate_hydrogen_price(
    const HydrogenRegion *r
) {
    double total_demand =
        r->industrial_demand +
        r->transport_demand +
        r->aerospace_demand;

    double effective_supply =
        r->hydrogen_inventory +
        r->production_capacity;

    if (effective_supply <= 0.0)
        return 100000.0;

    double scarcity =
        total_demand / effective_supply;

    /*
        Base commodity price.
        This is deliberately a game-economy abstraction,
        rather than a real-world hydrogen pricing model.
    */

    double base_price = 2500.0;

    double multiplier =
        1.0 + pow(scarcity, 1.7);

    /*
        Emergency scarcity premium.
    */

    if (r->hydrogen_inventory < r->minimum_reserve) {
        multiplier *= 1.5;
    }

    double price = base_price * multiplier;

    /*
        Prevent absurd values in the game economy.
    */

    if (price < 250.0)
        price = 250.0;

    if (price > 100000.0)
        price = 100000.0;

    return price;
}

double hydrogen_rarity(

    double demand =
        r->industrial_demand +
        r->transport_demand +
        r->aerospace_demand;

    double supply =
        r->hydrogen_inventory +
        r->production_capacity;

    if (supply <= 0.0)

 * scarcity = demand / available supply
 *
 * The result is transformed into a bounded
 * 0-1 rarity value for the game economy.
 */
double calculate_scarcity(
    double supply,
    double demand
) {
    if (supply <= 0.0)
        return 1.0;

    double ratio = demand / supply;

    return 1.0 - exp(-ratio);
}


/*
 * Price responds non-linearly to scarcity.
 */
double calculate_weapon_price(
    double base_price,
    double scarcity
) {
    double multiplier =
        1.0 + pow(scarcity * 10.0, 2.0);

    if (multiplier > 100.0)
        multiplier = 100.0;

    return base_price * multiplier;
}


/*
 * Update the market.
 */
void update_rare_weapon_market(
    RareWeaponMarket *market,
    double days
) {
    /*
     * Production adds to the global stock.
     */
    double production =
        market->production_rate * days;

    market->global_stock += production;

    /*
     * Demand removes stock.
     */
    double consumption =
        market->demand_rate * days;

    if (consumption > market->global_stock)
        consumption = market->global_stock;

    market->global_stock -= consumption;

    /*
     * Scarcity calculation.
     */
    market->scarcity =
        calculate_scarcity(
            market->global_stock,
            market->demand_rate
        );

    /*
     * Rarity is deliberately amplified so that
     * this remains the game's exceptional weapon.
     */
    market->rarity =
        pow(market->scarcity, 0.35);

    /*
     * Dynamic market price.
     */
    market->current_price =
        calculate_weapon_price(
            market->base_price,
            market->scarcity
        );

    market->available =
        market->global_stock > 0.0;
}

void airship_update(
    Airship *ship,
    Vector3 *position,
    double throttle,
    double climb_input,
    double dt
) {
    if (!ship->airborne)
        return;

    if (ship->hydrogen <= 0.0) {
        ship->speed *= 0.95;
        return;
    }

    /*
     * Abstract game movement model.
     */

    if (throttle < 0.0)
        throttle = 0.0;

    if (throttle > 1.0)
        throttle = 1.0;

    ship->speed =
        ship->max_speed * throttle;

    double heading_rad =
        ship->heading * PI / 180.0;

    position->x +=
        cos(heading_rad) *
        ship->speed * dt;

    position->y +=
        sin(heading_rad) *
        ship->speed * dt;

    /*
     * Altitude control.
     */

    position->z +=
        climb_input *
        ship->climb_rate *
        dt;

    if (position->z < 0.0)
        position->z = 0.0;

    /*
     * Hydrogen consumption.
     */

    double consumption =
        ship->hydrogen_consumption *
        throttle *
        dt;

    ship->hydrogen -= consumption;

    if (ship->hydrogen < 0.0)
        ship->hydrogen = 0.0;
}

    Train *train,
    Route *route,
    int current_minute
)

    if (train->state ==
        TRAIN_TERMINATED)

    TimetableStop *stop =
        get_current_stop(route, train);

    if (!stop)

    if (current_minute <
        stop->arrival_minute)

        train->state =
            TRAIN_RUNNING;

        return;

    if (current_minute >=
        stop->arrival_minute &&
        current_minute <
        stop->departure_minute)

        train->speed = 0.0;

        return;

        stop->departure_minute)

        train->current_stop++;

        if (train->current_stop >=
            route->stop_count)
        {
            train->state =
                TRAIN_TERMINATED;

3. Bandit AI
void update_bandit(
    RailwayBandit *bandit,
    RailwayTrain trains[],
    int train_count
) {
    if (!bandit->active)
        return;

    switch (bandit->state) {

        case BANDIT_IDLE:
            bandit->state =
                BANDIT_SCOUTING;
            break;

        case BANDIT_SCOUTING:

            /*
             * Find an attractive train.
             */

            for (int i = 0;
                 i < train_count;
                 i++) {

                if (!trains[i].moving &&
                    trains[i].freight_train) {

                    bandit->state =
                        BANDIT_TARGETING;

                    break;
                }
            }

            break;

        case BANDIT_TARGETING:

            /*
             * Game abstraction:
             * attempt to create a crime incident.
             */

            bandit->state =
                BANDIT_COMMITTING;

            break;

        case BANDIT_COMMITTING:

            bandit->stolen_value += 1000.0;

            bandit->notoriety += 5.0;

            bandit->state =
                BANDIT_ESCAPING;

            break;

        case BANDIT_ESCAPING:

            /*
             * Police AI takes over if detected.
             */

            break;

        case BANDIT_CAUGHT:
            bandit->active = false;
            break;
    }
}

double distance(
    double x1,
    double y1,
    double x2,
    double y2
) {
    double dx = x2 - x1;
    double dy = y2 - y1;

    return sqrt(dx * dx + dy * dy);
}


int find_nearest_police(
    TransportPolice police[],
    double x,
    double y
) {
    int best = -1;
    double best_distance = 1e30;

    for (int i = 0;
         i < MAX_POLICE;
         i++) {

        if (!police[i].active)
            continue;

        double d =
            distance(
                police[i].x,
                police[i].y,
                x,
                y
            );

        if (d < best_distance) {
            best_distance = d;
            best = i;
        }
    }

    return best;
}

Then dispatch:

void dispatch_police(
    TransportPolice police[],
    RailwayIncident *incident
) {
    int unit =
        find_nearest_police(
            police,
            incident->x,
            incident->y
        );

    if (unit < 0)
        return;

    police[unit].state =
        POLICE_RESPONDING;

    police[unit].target_bandit =
        incident->id;
}

void update_police(
    TransportPolice *police,
    RailwayBandit bandits[]
) {
    if (!police->active)
        return;

    if (police->target_bandit < 0)
        return;

    RailwayBandit *target =
        &bandits[police->target_bandit];

    switch (police->state) {

        case POLICE_PATROL:
            break;

        case POLICE_RESPONDING:

            police->state =
                POLICE_PURSUING;

            break;

        case POLICE_PURSUING:

            if (detect_bandit(
                    police,
                    target)) {

                police->state =
                    POLICE_ARRESTING;
            }

            break;

        case POLICE_ARRESTING:

            target->state =
                BANDIT_CAUGHT;

            police->state =
                POLICE_PATROL;

            police->target_bandit =
                -1;

            break;

        case POLICE_INVESTIGATING:
            break;
    }
}

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <math.h>
#include <stdbool.h>
#include <time.h>

#define MAX_COMMODITIES 32
#define MAX_REGIONS 32
#define MAX_CONSIGNMENTS 512
#define MAX_TRAIN_CARS 32

/* =========================================================
   COMMODITIES
   ========================================================= */

typedef enum {
    COMMODITY_LITHIUM,
    COMMODITY_HYDROGEN,
    COMMODITY_RARE_EARTHS,
    COMMODITY_WEAPONS,
    COMMODITY_STEEL,
    COMMODITY_CEMENT,
    COMMODITY_FOOD,
    COMMODITY_ELECTRONICS,
    COMMODITY_FUEL,
    COMMODITY_MACHINE_PARTS
} CommodityType;


typedef struct {
    CommodityType type;

    char name[64];

    /*
     * Base economic value.
     * Game abstraction rather than real-world pricing.
     */
    double base_price;

    /*
     * Total quantity currently available
     * within the region.
     */
    double stock;

    /*
     * Desired inventory level.
     */
    double target_stock;

    /*
     * Daily consumption.
     */
    double daily_demand;

    /*
     * Local production.
     */
    double daily_production;

    /*
     * External imports arriving by rail/ship.
     */
    double daily_imports;

    /*
     * Economic scarcity: 0 = abundant,
     * 1 = extreme scarcity.
     */
    double scarcity;

    double price;
} Commodity;


/* =========================================================
   REGION
   ========================================================= */

typedef struct {

    int id;

    char name[64];

    Commodity commodities[MAX_COMMODITIES];

    double economic_activity;

    double import_dependency;

} FreightRegion;


/* =========================================================
   CONSIGNMENT
   ========================================================= */

typedef enum {
    FREIGHT_PLANNED,
    FREIGHT_LOADING,
    FREIGHT_IN_TRANSIT,
    FREIGHT_DELIVERED,
    FREIGHT_LOST,
    FREIGHT_CANCELLED
} FreightStatus;


typedef struct {

    int id;

    CommodityType commodity;

    double quantity;

    double cargo_value;

    int origin_region;
    int destination_region;

    int origin_terminal;
    int destination_terminal;

    double distance_km;

    double transport_cost;

    double expected_revenue;

    double profit;

    FreightStatus status;

} FreightConsignment;


/* =========================================================
   RAIL FREIGHT TRAIN
   ========================================================= */

typedef struct {

    int id;

    char name[64];

    double capacity_tonnes;

    double loaded_tonnes;

    double operating_cost_per_km;

    double position;

    double route_length;

    bool active;

    FreightConsignment *cargo[MAX_TRAIN_CARS];

    int cargo_count;

} FreightTrain;


/* =========================================================
   COMMODITY INITIALISATION
   ========================================================= */

void initialise_commodity(
    Commodity *c,
    CommodityType type,
    const char *name,
    double base_price,
    double stock,
    double target_stock,
    double demand,
    double production
) {
    c->type = type;

    snprintf(
        c->name,
        sizeof(c->name),
        "%s",
        name
    );

    c->base_price = base_price;
    c->stock = stock;
    c->target_stock = target_stock;
    c->daily_demand = demand;
    c->daily_production = production;

    c->daily_imports = 0.0;

    c->scarcity = 0.0;
    c->price = base_price;
}


/* =========================================================
   SCARCITY ENGINE
   ========================================================= */

double calculate_scarcity(
    double stock,
    double target_stock
) {
    if (target_stock <= 0.0)
        return 0.0;

    /*
     * Stock above target = abundance.
     */

    if (stock >= target_stock)
        return 0.0;

    double deficit =
        1.0 -
        (stock / target_stock);

    if (deficit < 0.0)
        deficit = 0.0;

    if (deficit > 1.0)
        deficit = 1.0;

    return deficit;
}


/* =========================================================
   PRICE ENGINE
   ========================================================= */

double calculate_price(
    double base_price,
    double scarcity,
    double economic_activity
) {
    /*
     * Non-linear price response.
     */

    double scarcity_multiplier =
        1.0 +
        pow(scarcity, 1.8) * 8.0;

    double demand_multiplier =
        0.75 +
        economic_activity * 0.5;

    double price =
        base_price *
        scarcity_multiplier *
        demand_multiplier;

    return price;
}


/* =========================================================
   UPDATE COMMODITY MARKET
   ========================================================= */

void update_commodity_market(
    FreightRegion *region,
    Commodity *commodity,
    double days
) {
    /*
     * Local production.
     */

    commodity->stock +=
        commodity->daily_production * days;

    /*
     * Imports.
     */

    commodity->stock +=
        commodity->daily_imports * days;

    /*
     * Consumption.
     */

    commodity->stock -=
        commodity->daily_demand *
        region->economic_activity *
        days;

    if (commodity->stock < 0.0)
        commodity->stock = 0.0;

    /*
     * Recalculate scarcity.
     */

    commodity->scarcity =
        calculate_scarcity(
            commodity->stock,
            commodity->target_stock
        );

    /*
     * Recalculate price.
     */

    commodity->price =
        calculate_price(
            commodity->base_price,
            commodity->scarcity,
            region->economic_activity
        );
}


/* =========================================================
   IMPORT DEMAND
   ========================================================= */

double calculate_import_requirement(
    const Commodity *commodity
) {
    if (commodity->stock >=
        commodity->target_stock)
        return 0.0;

    return
        commodity->target_stock -
        commodity->stock;
}


/* =========================================================
   TRANSPORT COST
   ========================================================= */

double calculate_transport_cost(
    double distance_km,
    double tonnes,
    double cost_per_tonne_km
) {
    return
        distance_km *
        tonnes *
        cost_per_tonne_km;
}


/* =========================================================
   CREATE CONSIGNMENT
   ========================================================= */

FreightConsignment create_consignment(
    int id,
    CommodityType commodity,
    double quantity,
    double price,
    int origin,
    int destination,
    double distance
) {
    FreightConsignment c = {0};

    c.id = id;

    c.commodity = commodity;

    c.quantity = quantity;

    c.cargo_value =
        quantity * price;

    c.origin_region = origin;
    c.destination_region = destination;

    c.distance_km = distance;

    /*
     * Abstract railway logistics cost.
     */

    c.transport_cost =
        calculate_transport_cost(
            distance,
            quantity,
            0.08
        );

    c.expected_revenue =
        c.cargo_value;

    c.profit =
        c.expected_revenue -
        c.transport_cost;

    c.status =
        FREIGHT_PLANNED;

    return c;
}


/* =========================================================
   LOAD TRAIN
   ========================================================= */

bool load_consignment(
    FreightTrain *train,
    FreightConsignment *cargo
) {
    if (cargo->status !=
        FREIGHT_PLANNED)
        return false;

    if (train->loaded_tonnes +
        cargo->quantity >
        train->capacity_tonnes)
        return false;

    if (train->cargo_count >=
        MAX_TRAIN_CARS)
        return false;

    train->cargo[
        train->cargo_count++
    ] = cargo;

    train->loaded_tonnes +=
        cargo->quantity;

    cargo->status =
        FREIGHT_LOADING;

    return true;
}


/* =========================================================
   DEPART TRAIN
   ========================================================= */

void depart_freight_train(
    FreightTrain *train
) {
    if (!train->active)
        return;

    for (int i = 0;
         i < train->cargo_count;
         i++) {

        FreightConsignment *cargo =
            train->cargo[i];

        if (cargo == NULL)
            continue;

        cargo->status =
            FREIGHT_IN_TRANSIT;
    }

    printf(
        "TRAIN %s departed with %.1f tonnes\n",
        train->name,
        train->loaded_tonnes
    );
}


/* =========================================================
   MOVE TRAIN
   ========================================================= */

void update_freight_train(
    FreightTrain *train,
    double speed,
    double hours
) {
    if (!train->active)
        return;

    train->position +=
        speed * hours;

    if (train->position >=
        train->route_length) {

        train->position =
            train->route_length;

        for (int i = 0;
             i < train->cargo_count;
             i++) {

            FreightConsignment *cargo =
                train->cargo[i];

            if (cargo != NULL)
                cargo->status =
                    FREIGHT_DELIVERED;
        }

        train->active = false;

        printf(
            "TRAIN %s ARRIVED\n",
            train->name
        );
    }
}


/* =========================================================
   IMPORT INTO REGION
   ========================================================= */

void deliver_import(
    FreightRegion *region,
    FreightConsignment *cargo
) {
    if (cargo->status !=
        FREIGHT_DELIVERED)
        return;

    Commodity *commodity =
        &region->commodities[
            cargo->commodity
        ];

    commodity->stock +=
        cargo->quantity;

    commodity->daily_imports +=
        cargo->quantity;

    printf(
        "%s: %.1f tonnes of %s delivered\n",
        region->name,
        cargo->quantity,
        commodity->name
    );
}


/* =========================================================
   MARKET REPORT
   ========================================================= */

void print_market(
    FreightRegion *region,
    CommodityType type
) {
    Commodity *c =
        &region->commodities[type];

    printf(
        "\n=== %s MARKET ===\n",
        c->name
    );

    printf(
        "Stock:       %.1f tonnes\n",
        c->stock
    );

    printf(
        "Demand:      %.1f tonnes/day\n",
        c->daily_demand
    );

    printf(
        "Scarcity:    %.2f\n",
        c->scarcity
    );

    printf(
        "Price:       £%.2f / tonne\n",
        c->price
    );
}

void update_network_operation(

        return;

        operation->intensity *

        0.1;

        generated_load;

    target->current_load +=
        generated_load;

    operation->duration +=
        seconds;

    /*
     * Detection increases with

    operation->detection +=
        operation->intensity *

            target->name

     * Detection threshold.

    if (effective >= 75.0)

    if (effective >= 55.0)

        return;

        dc->incident_response *
        seconds *
        0.01;

    event->progress +=
        response_power;

    if (event->progress >=
        event->severity * 100.0) {

        event->resolved = true;
        event->active = false;

        printf(

    if (dc->heat < 10.0)
        dc->lockdown = false;

    if (dc->heat < 0.0)

}

Power simulation
void update_power(
    DataCentre *dc,
    double seconds
) {
    double load = 0.0;

    for (int i = 0;
         i < dc->rack_count;
         i++) {

        ServerRack *rack =
            &dc->racks[i];

        if (!rack->operational)
            continue;

        load +=
            rack->power_consumption *
            (rack->current_compute /
             rack->compute_capacity);
    }

    dc->power.load = load;

    if (dc->power.grid_online &&
        load <= dc->power.grid_capacity) {

        dc->power.generators_online =
            false;

        return;
    }

    /*
     * Automatic backup power.
     */

    if (dc->power.generators_online &&
        load <=
        dc->power.generator_capacity) {

        return;
    }

    /*
     * Battery support.
     */

    if (dc->power.current_battery > 0.0) {

        dc->power.current_battery -=
            load * seconds / 3600.0;

        if (dc->power.current_battery < 0.0)
            dc->power.current_battery = 0.0;

        return;
    }

    /*
     * Power shortage:
     * reduce available compute.
     */

    dc->current_compute *= 0.90;
}
Cooling

Every rack generates abstract heat, and the cooling system has to remove it.

void update_cooling(
    DataCentre *dc
) {
    double heat = 0.0;

    for (int i = 0;
         i < dc->rack_count;
         i++) {

        ServerRack *rack =
            &dc->racks[i];

        if (!rack->operational)
            continue;

        heat +=
            rack->heat_output *
            (rack->current_compute /
             rack->compute_capacity);
    }

    dc->cooling.current_load =
        heat;

    if (heat >
        dc->cooling.total_capacity) {

        /*
         * Thermal throttling.
         */

        double ratio =
            dc->cooling.total_capacity /
            heat;

        if (ratio < 0.25)
            ratio = 0.25;

        dc->current_compute *= ratio;
    }
}

Dynamic GPU pricing

The GPU price should move according to scarcity + compute demand + power availability + capital conditions.

double calculate_gpu_price(
    GPUAsset *gpu,
    GPUMarket *market
) {
    double supply =
        gpu->supply;

    double demand =
        gpu->demand;

    if (supply < 1.0)
        supply = 1.0;

    double scarcity =
        demand / supply;

    /*
     * Price rises non-linearly
     * when demand exceeds supply.
     */

    double multiplier =
        0.65 +
        pow(scarcity, 1.35);

    /*
     * Clamp extreme values.
     */

    if (multiplier < 0.40)
        multiplier = 0.40;

    if (multiplier > 5.0)
        multiplier = 5.0;

    gpu->market_price =
        gpu->purchase_price *
        multiplier;

    return gpu->market_price;
}

bool buy_gpu(
    GPUMarket *market,
    GPUVault *vault,
    GPUType type,
    int quantity
) {
    GPUAsset *gpu =
        &market->gpu[type];

    double cost =
        gpu->market_price *
        quantity;

    if (vault->capital < cost)
        return false;

    if (gpu->supply < quantity)
        return false;

    vault->capital -= cost;

    gpu->supply -= quantity;

    /*
     * Add compute capacity.
     */

    vault->gpu_count += quantity;

    vault->total_compute +=
        gpu->compute_units *
        quantity;

    vault->total_memory +=
        gpu->memory_gb *
        quantity;

    vault->current_power_mw +=
        gpu->power_kw *
        quantity /
        1000.0;

    return true;
}

    double y;

    double weight;

    double health;
    double energy;
    double hunger;

    double fear;

    /* Territory */
    double territory_x;
    double territory_y;
    double territory_radius;

    /* Perception */
    double vision_range;
    double hearing_range;

    /* Behaviour */
    double aggression;
    double territoriality;

    bool female;
    bool has_nest;
    bool has_young;

    GatorState state;

    bool alive;
} Alligator;


/* =========================================================
   DISTANCE
   ========================================================= */

double distance(
    double x1,
    double y1,
    double x2,
    double y2
) {
    double dx = x2 - x1;
    double dy = y2 - y1;

    return sqrt(
        dx * dx +
        dy * dy
    );
}


/* =========================================================
   TERRITORY
   ========================================================= */

bool inside_territory(
    Alligator *gator,
    double x,
    double y
) {
    double d =
        distance(
            gator->territory_x,
            gator->territory_y,
            x,
            y
        );

    return d <=
           gator->territory_radius;
}


/* =========================================================
   PLAYER / NPC PERCEPTION
   ========================================================= */

double threat_level(
    Alligator *gator,
    double player_x,
    double player_y,
    bool player_running,
    bool player_near_nest
) {
    double d =
        distance(
            gator->x,
            gator->y,
            player_x,
            player_y
        );

    if (d > gator->vision_range)
        return 0.0;

    /*
     * Being close increases perceived threat.
     */

    double threat =
        1.0 -
        d / gator->vision_range;

    /*
     * Running / sudden movement.
     */

    if (player_running)
        threat += 0.20;

    /*
     * Nest protection is a major modifier.
     */

    if (player_near_nest &&
        gator->female &&
        gator->has_nest) {

        threat += 0.50;
    }

    return threat;

        gator->id

        GATOR_WARNING;
}


/* =========================================================
   DEFENSIVE RESPONSE
   ========================================================= */

void defensive_response(
    Alligator *gator,
    double threat
) {
    /*
     * Most encounters should resolve
     * through warning or retreat rather
     * than immediate aggression.
     */

    if (threat < 0.35) {

        gator->state =
            GATOR_ALERT;

        return;
    }

    if (threat < 0.65) {

        issue_warning(gator);

        return;
    }

    /*
     * High threat.
     */

    if (gator->fear > 60.0) {

        gator->state =
            GATOR_RETREATING;

        return;
    }

    /*
     * Nesting females are strongly
     * defensive.
     */

    if (gator->female &&
        gator->has_nest) {

        gator->state =
            GATOR_DEFENDING;

        return;
    }

    /*
     * Otherwise retreat first.
     */

    gator->state =
        GATOR_RETREATING;
}


/* =========================================================
   ALLIGATOR AI
   ========================================================= */

void update_alligator(
    Alligator *gator,
    double player_x,
    double player_y,
    bool player_running,
    bool player_near_nest,
    double delta_time
) {
    if (!gator->alive)
        return;

    double threat =
        threat_level(
            gator,
            player_x,
            player_y,
            player_running,
            player_near_nest
        );

    switch (gator->state) {

        case GATOR_IDLE:

            if (gator->hunger > 70.0)
                gator->state =
                    GATOR_FORAGING;

            else if (threat > 0.15)
                gator->state =
                    GATOR_ALERT;

            break;


        case GATOR_BASKING:

            /*
             * Slowly recover energy.
             */

            gator->energy +=
                2.0 * delta_time;

            if (gator->energy > 100.0)
                gator->energy = 100.0;

            if (threat > 0.15)
                gator->state =
                    GATOR_ALERT;

            break;


        case GATOR_SWIMMING:

            gator->energy -=
                1.0 * delta_time;

            if (threat > 0.25)
                gator->state =
                    GATOR_ALERT;

            break;


        case GATOR_FORAGING:

            gator->hunger -=
                5.0 * delta_time;

            if (gator->hunger < 20.0)
                gator->hunger = 20.0;

            if (threat > 0.30)
                gator->state =
                    GATOR_ALERT;

            break;


        case GATOR_ALERT:

            defensive_response(
                gator,
                threat
            );

            break;


        case GATOR_WARNING:

            /*
             * If threat decreases,
             * return to normal behaviour.
             */

            if (threat < 0.25) {

                gator->state =
                    GATOR_IDLE;
            }

            /*
             * Escalate only if the player
             * continues approaching.
             */

            else if (threat > 0.75) {

                gator->state =
                    GATOR_DEFENDING;
            }

            break;


        case GATOR_DEFENDING:

            /*
             * Defensive state.
             * Return to normal once
             * perceived threat leaves
             * the territory.
             */

            if (threat < 0.40) {

                gator->state =
                    GATOR_RETREATING;
            }

            break;


        case GATOR_RETREATING:

            /*
             * Move back toward water /
             * safe territory.
             */

            gator->fear -=
                5.0 * delta_time;

            if (gator->fear < 20.0)
                gator->fear = 20.0;

            if (threat < 0.15) {

                gator->state =
                    GATOR_SWIMMING;
            }

            break;


        case GATOR_NESTING:

            if (gator->female &&
                gator->has_nest &&
                threat > 0.25) {

                gator->state =
                    GATOR_DEFENDING;
            }

            break;


        case GATOR_FLEEING:

            if (threat < 0.10)
                gator->state =
                    GATOR_SWIMMING;

            break;
    }

void update_population(
    AlligatorPopulation *population,
    double days
) {
    int births =
        (int)(
            population->population *
            population->birth_rate *
            days
        );

    int deaths =
        (int)(
            population->population *
            population->mortality_rate *
            days

        births - deaths;

}

/* =========================================================
   DISTANCE
   ========================================================= */

double distance(
    double x1,
    double y1,
    double x2,
    double y2
) {
    double dx = x2 - x1;
    double dy = y2 - y1;

    return sqrt(dx * dx + dy * dy);
}


/* =========================================================
   TERRITORY
   ========================================================= */

bool inside_tiger_territory(
    Tiger *tiger,
    double x,
    double y
) {
    return distance(
        tiger->territory_x,
        tiger->territory_y,
        x,
        y
    ) <= tiger->territory_radius;
}


/* =========================================================
   PERCEPTION
   ========================================================= */

double calculate_threat(
    Tiger *tiger,
    double x,
    double y,
    bool moving_fast,
    bool near_cubs
) {
    double d =
        distance(
            tiger->x,
            tiger->y,
            x,
            y
        );

    if (d > tiger->vision_range)
        return 0.0;

    double threat =
        1.0 -
        d / tiger->vision_range;

    if (moving_fast)
        threat += 0.20;

    if (near_cubs &&
        tiger->female &&
        tiger->has_cubs)
        threat += 0.50;

    return threat;
}


/* =========================================================
   FIND PREY
   ========================================================= */

int find_nearest_prey(
    Tiger *tiger,
    Prey prey[],
    int count
) {
    int best = -1;
    double best_distance = 1e30;

    for (int i = 0; i < count; i++) {

        if (!prey[i].alive)
            continue;

        double d =
            distance(
                tiger->x,
                tiger->y,
                prey[i].x,
                prey[i].y
            );

        if (d <= tiger->vision_range &&
            d < best_distance) {

            best_distance = d;
            best = i;
        }
    }

    return best;
}


/* =========================================================
   TIGER AI
   ========================================================= */

void update_tiger(
    Tiger *tiger,
    Prey prey[],
    int prey_count,
    double player_x,
    double player_y,
    bool player_running,
    bool player_near_cubs,
    double delta_time
) {
    if (!tiger->alive)
        return;

    double player_threat =
        calculate_threat(
            tiger,
            player_x,
            player_y,
            player_running,
            player_near_cubs
        );

    /*
     * Energy naturally declines.
     */

    tiger->energy -=
        0.5 * delta_time;

    if (tiger->energy < 0.0)
        tiger->energy = 0.0;

    /*
     * Hunger increases over time.
     */

    tiger->hunger +=
        1.0 * delta_time;

    if (tiger->hunger > 100.0)
        tiger->hunger = 100.0;


    switch (tiger->state) {

        case TIGER_RESTING:

            if (player_threat > 0.25) {

                tiger->state =
                    TIGER_ALERT;

            } else if (tiger->hunger > 65.0) {

                tiger->state =
                    TIGER_PATROLLING;
            }

            break;


        case TIGER_PATROLLING:

            /*
             * Look for prey.
             */

            tiger->target_prey =
                find_nearest_prey(
                    tiger,
                    prey,
                    prey_count
                );

            if (tiger->target_prey >= 0) {

                tiger->state =
                    TIGER_STALKING;

            } else if (player_threat > 0.30) {

                tiger->state =
                    TIGER_ALERT;
            }

            break;


        case TIGER_STALKING:

            /*
             * Abstract hunting state.
             */

            if (tiger->target_prey < 0) {

                tiger->state =
                    TIGER_PATROLLING;

                break;
            }

            if (!prey[
                    tiger->target_prey
                ].alive) {

                tiger->target_prey = -1;

                tiger->state =
                    TIGER_PATROLLING;

                break;
            }

            /*
             * Avoid hunting while severely
             * exhausted.
             */

            if (tiger->energy < 20.0) {

                tiger->state =
                    TIGER_RESTING;

                break;
            }

            tiger->state =
                TIGER_HUNTING;

            break;


        case TIGER_HUNTING:

            if (tiger->target_prey < 0) {

                tiger->state =
                    TIGER_PATROLLING;

                break;
            }

            /*
             * Fictional game resolution.
             */

            if (tiger->energy > 30.0) {

                prey[
                    tiger->target_prey
                ].health -=
                    20.0 * delta_time;

                tiger->energy -=
                    5.0 * delta_time;
            }

            if (prey[
                    tiger->target_prey
                ].health <= 0.0) {

                prey[
                    tiger->target_prey
                ].alive = false;

                tiger->hunger -= 70.0;

                if (tiger->hunger < 0.0)
                    tiger->hunger = 0.0;

                tiger->state =
                    TIGER_EATING;
            }

            break;


        case TIGER_ALERT:

            /*
             * Prefer avoidance where possible.
             */

            if (player_threat < 0.20) {

                tiger->state =
                    TIGER_PATROLLING;

            } else if (player_threat > 0.70) {

                if (tiger->female &&
                    tiger->has_cubs) {

                    tiger->state =
                        TIGER_DEFENDING;

                } else {

                    tiger->state =
                        TIGER_FLEEING;
                }
            }

            break;


        case TIGER_DEFENDING:

            /*
             * Defensive state around cubs.
             */

            if (player_threat < 0.35) {

                tiger->state =
                    TIGER_FLEEING;
            }

            break;


        case TIGER_FLEEING:

            tiger->fear -=
                4.0 * delta_time;

            if (player_threat < 0.10) {

                tiger->state =
                    TIGER_RESTING;
            }

            break;


        case TIGER_EATING:

            tiger->energy +=
                4.0 * delta_time;

            tiger->hunger -=
                8.0 * delta_time;

            if (tiger->hunger < 20.0)
                tiger->hunger = 20.0;

            if (tiger->energy > 70.0)
                tiger->state =
                    TIGER_RESTING;

            break;


        case TIGER_WARNING:

            if (player_threat < 0.20)
                tiger->state =
                    TIGER_FLEEING;

            break;


        case TIGER_INJURED:

            tiger->energy -=
                0.2 * delta_time;

            if (tiger->energy <= 0.0)
                tiger->alive = false;

            break;
    }
}

And importantly, Khan's "superhuman" quality doesn't have to mean magical powers. In gameplay terms it can mean extraordinary perception, persistence, intelligence, memory, stealth and physical attributes—making him feel almost superhuman while remaining recognisably an animal.

That also makes him much more interesting as a GTA-style antagonist: the player can't simply clear an area of ordinary enemies and move on. Khan is a persistent character who owns part of the world, remembers encounters and can reappear much later in the story.
